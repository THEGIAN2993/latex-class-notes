\begin{definition}
    The \textui{completion} of a metric space $(X,d)$ consists of:
        \begin{enumerate}[label = (\arabic*),itemsep=1pt,topsep=3pt]
            \item A complete metric space $(Z,\rho)$;
            \item An isometry $\iota:X \hookrightarrow Z$;
            \item The image of $X$ is dense in $Z$; i.e., $\overline{\iota(X)}^\rho = Z$.
        \end{enumerate}
\end{definition}
