\begin{problem}[F25, P3]
Let $R$ be an integral domain. Suppose that $a$ and $b$ are non-associate irreducible elements in $R$, and the ideal $(a,b)$ generated by $a$ and $b$ is a proper ideal. Show that $R$ is not a principal ideal domain.
\end{problem}
    {\color{blue} \begin{proof}
        Suppose there exists an element $c \in R$ such that $(a,b) = (c)$. Then $a = cr$ and $b = cs$ for some $s,r \in \bfZ$. Since $a$ and $b$ are irreducible, either $c$ is a unit or $s$ and $r$ are units. If $c$ is a unit, then $(a,b) = R$, contradicting the fact that $(a,b)$ is a proper ideal. Hence it must be the case that $r$ and $s$ are units. This implies that $ar^{-1} = c = bs^{-1}$. Rearranging yields $ar^{-1}s = b$. This is a contradiction, as $r^{-1}s$ is a unit, and by assumption $a$ and $b$ were not associates. Thus $R$ is not a principal ideal domain.
    \end{proof}}

\begin{problem}[S23, P3]
Suppose $R$ is a PID. Prove that an ideal $I \subset R$ is maximal if and only if $I = (p)$ for a prime $p \in R$.
\end{problem}
    {\color{blue} \begin{proof}
        Let $I$ be maximal. Since $R$ is a PID, $I = (x)$ for some $x \in R$. If $x = ab$, then $(x) \subseteq (a)$. Hence $(a) = (x)$ or $(a) = R$ by the maximality of $(x)$. This means that $a$ is a unit or an associate of $x$. It must be the case that $x$ is irreducible. In a PID, irreducible elements are prime, so $x$ is a prime element and $I = (x) = (p)$ for some prime $p$.

        ($\Leftarrow$) Let $(m)$ be any ideal containing $(p)$. Then $p \in (m)$, so $p = rm$ for some $r \in R$. Since $(p)$ is a prime ideal and $rm \in (p)$, either $r \in (p)$ or $m \in (p)$. If $m \in (p)$ then $(m) = (p)$. If $r \in (p)$, then $r = ps$ for some $s \in \bfZ$. It follows that $p = rm = psm$, hence $sm = 1$. Since $m$ is a unit, $(m) = R$. Thus $(p)$ is maximal.
    \end{proof}}

\begin{problem}[F17, P3]
Prove that every Euclidean domain is a principal ideal domain.
\end{problem}
    {\color{blue} \begin{proof}
        Let $R$ be a Euclidean domain. Let $I \subset R$ be any ideal. Pick $x \in I$ with minimal norm (by the Well-ordering principle, such an $x$ exists since $\left\{\left|r\right| \mid r \in I\right\} \subset \bfN$). Clearly $(x) \subseteq I$, as $I$ is closed under multiplication by elements of $R$. Let $y \in I$. Write $y = xq + r$, where $q,r \in \bfZ$ such that $|r| < |x|$ or $r = 0$. Since $x$ was chosen to have minimal norm, it must be the case that $r = 0$. Hence $y = xq$, which gives $y \in (x)$. Thus $I = (x)$, establishing $R$ as a PID.
    \end{proof}}

\begin{problem}[W22, P4]
    Let $\bfZ_2 = \{0,1\}$ be the field of 2 elements. The quotient ring $\bfZ_2[x]/(x^3 + x + 1)$ be the natural projection.
    \begin{enumerate}[label = (\alph*),itemsep=1pt,topsep=3pt]
        \item Write down a set of 8 distinct coset representatives for the elements of this field.
        \item Determine the multiplicative inverse of $\pi(x)$ in terms of your coset representatives.
    \end{enumerate}
\end{problem}
    {\color{blue} \begin{proof}
        (a) Note that the coset representatives will be of the form $ax^2 + bx+c$ for $a,b,c \in \{0,1\}$. The coset representatives for the elements of this field are:
            \begin{equation*}
            \begin{split}
                \bigl\{[0],[1],[x],[x+1],[x^2],[x^2+1],[x^2+x],[x^2+x+1]\bigr\}.
            \end{split}
            \end{equation*}

        (b) Note that $[x^3 + x + 1] = [0]$, hence $[x^3] = [-x-1] = [x+1]$. It follows that $[x][x^2 + 1] = [x^3 + x] = [1]$, hence the inverse of $[x]$ is $[x^2 + 1]$.
    \end{proof}}
