\begin{exercise}
	\phantom{a}
	\mbox{}
	\begin{enumerate}[label = (\alph*),itemsep=1pt,topsep=3pt]
		\item Let $E$ be an $n \times n$ matrix over $\bfC$ such that $E \neq I_n$, $E \neq 0$, and $E^2 = E$. Show that there is an $n \times n$ matrix $T$ such that:
			\[
				E = T^{-1} \begin{bmatrix} I_k & 0 \\ 0 & 0 \end{bmatrix} T
			\]
		for some $k$.

		\item Let $G$ be a group and $X$ be a function mapping $G$ into the set of (possible singular) $n \times n$ matrices over $\bfC$ such that $X(\epsilon) \neq 0$ and $X(gh) = X(g)X(h)$ for all $g,h \in G$. Show that there is an $n \times n$ matrix $T$ and a matrix representation $Y$ of $G$ such that:
			\[
				X(g) = T^{-1} \begin{bmatrix} Y(g) & 0 \\ 0 & 0  \end{bmatrix} T
			\] 
			for all $g \in G$.
	\end{enumerate}
\end{exercise}
	\begin{proof}
		(a) Since $E$ is a projection, its eigenvalues are $\lambda = 0$ and $\lambda =1$. Let $E_0 := \ker(E)$ and $E_1 := \ker(E - I_n)$ denote the eigenspaces of $\lambda = 0$ and $\lambda  = 1$. Note that $Ev \in E_1$, as:
		\begin{align*}
			(E-I_n)(Ev)
			&= E^2v - Ev\\
			&= 0.
		\end{align*}
		Similarly, $v-Ev \in E_0$ since:
		\begin{align*}
			E(v-Ev)
			&= Ev - E^2v \\
			&= 0.
		\end{align*}
		Since every $v \in V$ can be written as $(v - Ev) + Ev$, this means $V = E_0 + E_1$. If $x \in E_0 \cap E_1$, then $Ex = 0$ and $Ex = x$. It must be the case that $x = 0$, hence $V = E_0 \oplus E_1$. Since $V$ is the direct sum of eigenspaces, $E$ is diagonalizable. Assuming that $\dim(E_1) = k$ and $\dim(E_0) = n-k$, there exists a matrix $T$ such that $E = T^{-1}\begin{bmatrix} I_k & 0 \\ 0 & 0 \end{bmatrix} T$.

		(b) Note that $X(\epsilon)^2 = X(\epsilon)X(\epsilon) = X(\epsilon)$. Hence by part (a) there exists an $n \times n$ matrix $T$ such that:
		\[
			X(g) = T^{-1}\begin{bmatrix} I_k & 0 \\ 0 & 0 \end{bmatrix} T.
		\]
		Note that:
		\[
			T\,X(g)\,T^{-1} = (T\,X(g)\,T^{-1})(T\,X(\epsilon)\,T^{-1}) = (T\,X(g)\,T^{-1}) \begin{bmatrix} I_k & 0 \\ 0 & 0 \end{bmatrix}.
		\]
		Similarly:
		\[
			T\,X(g)\,T^{-1} = (T\,X(\epsilon)\,T^{-1})(T\,X(g)\,T^{-1}) = \begin{bmatrix} I_k & 0 \\ 0 & 0 \end{bmatrix} (T\,X(g)\,T^{-1}).
		\]
		Since $(T\,X(g)\,T^{-1}) \begin{bmatrix} I_k & 0 \\ 0 & 0 \end{bmatrix} = \begin{bmatrix} I_k & 0 \\ 0 & 0 \end{bmatrix} (T\,X(g)\,T^{-1})$, it must be the case that $T\,X(g)\,T^{-1} = \begin{bmatrix} Y(g) & 0 \\ 0 & 0 \end{bmatrix} $ for some $Y(g) \in \Mat_k(\bfC)$. It remains to show that $Y$ is a representation. Note that:
		\[
			\begin{bmatrix} Y(\epsilon) & 0 \\ 0 & 0 \end{bmatrix} = T\,X(\epsilon)\,T^{-1} = \begin{bmatrix} I_k & 0 \\ 0 & 0 \end{bmatrix}.
		\] 
		Thus $Y(\epsilon) = I_k$. Similarly:
		\begin{align*}
			\begin{bmatrix} Y(gh) & 0 \\ 0 & 0 \end{bmatrix} 
			&= T\,X(gh)\,T^{-1} \\
			&= (T\,X(g)\,T^{-1})(T\,X(h)\,T^{-1}) \\
			&= \begin{bmatrix} Y(g) & 0 \\ 0 & 0 \end{bmatrix} \begin{bmatrix} Y(h) & 0 \\ 0 & 0 \end{bmatrix} \\
			&= \begin{bmatrix} Y(g)Y(h) & 0 \\ 0 & 0 \end{bmatrix}.
		\end{align*}
		Thus $Y(gh) = Y(g)Y(h)$. It remains to show that $\Image Y = GL_n(\bfC)$. Indeed, for any $Y(g) \in \Image Y$ we have $Y(\epsilon) = Y(g g^{-1}) = Y(g) Y(g^{-1})$, hence $Y(g)$ does have an inverse. Thus, we've shown that there exists an $n \times n$ matrix $T$ and matrix representation $Y$ of $G$ such that:
		\[
			X(g) = T^{-1} \begin{bmatrix} Y(g) & 0 \\ 0 & 0 \end{bmatrix} T\qedhere
		\]
	\end{proof}

\begin{exercise}
	For $\sigma = \sigma_1\ldots \sigma_n \in S_n$, written in one line notation, $\inv \sigma$ is the number of pairs $i < j$ such that $\sigma_i > \sigma_j$. Define $\sign \sigma := (-1)^{\inv \sigma}$. Show that switching the position of any two integers in $\sigma$ changes $\inv \sigma$ by an odd number. This fact implies:
	\[
		\sign(\sigma \tau) = \sign \sigma \, \sign \tau,
	\] 
	where $\tau$ is any transposition. Since any permutation can be written as a product of transpositions, this means that the sign representation $X(\sigma) = \sign (\sigma)$ is indeed a matrix representation of $S_n$.
\end{exercise}
	\begin{proof}
		Let $\tau$ be obtained from $\sigma$ by swapping the entries in positions $p < q$. After the swap, suppose $\sigma_p < \sigma_q$. Let $m = \card \{r: p < r < q \text{ and }\sigma_p < \sigma_r < \sigma_q\}$. Any inversions involving neither $p$ nor $q$ is unchanged, hence the total number of inversions is unchanged. Entries outside the interval from $p$ to $q$ also leave the total number of inversions unchanged. The pair $(p,q)$ was not an inversion in $\sigma$, but becomes one in $\tau$, contributing $+1$. Now consider $p < r < q$. If $\sigma_p < \sigma_r < \sigma_q$, then after the swap $(p,r)$ and $(r,q)$ are inversions. Thus each $r$ contributes $+2$. If $\sigma_r < \sigma_p$ or $\sigma_r > \sigma_q$, then in either case the pairs contribute one inversion both before and after the swap, so there is no change. Thus:
		\[
			\inv(\tau) - \inv(\sigma) = 1 + 2m,
		\] 
		which is odd. If $\sigma_p > \sigma_q$, the same argument gives $-(1+2m)$ which is also odd. 

		If $\sigma \in S_n$ and $\tau$ is a transposition, then:{\color{red} DNF}
	\end{proof}

\begin{exercise}
	Let $X:H \rightarrow \GL_n(\bfC)$ be a representation and $H$ a subgroup of $G$. Show that the induced representation $X\uair_H^G$ is indeed a representation. asf
\end{exercise}
	\begin{proof}
		Suppose $g_1H,\ldots ,g_kH$ are the left cosets of $H$ in $G$. We have:
		\[
			\bigl( X \uair_H^G (\epsilon) \bigr)_{ij} =
			\begin{cases}
				X(g_i^{-1}g_j), & g_j \in g_iH \\
				0, & g_j \not\in g_iH
			\end{cases}.
		\] 
		Since the left cosets of $H$ are disjoint, for all $i \neq j$ we have $g_j \not\in g_iH$. Hence:
		\[
			\bigl( X \uair_H^G (\epsilon) \bigr)_{ij} =
			\begin{cases}
				X(\epsilon), & i = j \\\
				0, & i \neq j
			\end{cases}.
		\]
		Since $X$ is a representation, $X(\epsilon) = I_n$. Thus:
		\[
			X\uair_H^G(\epsilon) = I_{nk}.
		\] 
		Now for any $x,y \in G$, we have:
		\begin{align*}
			\bigl( X\uair_H^G(x) X\uair_H^G(y) \bigr)_{ij}
			&= \sum_{\alpha = 1}^{k} \bigl( X\uair_H^G(x) \bigr)_{i\alpha} \bigl( X\uair_H^G(y) \bigr)_{\alpha j} \\
			&= \sum_{\alpha = 1}^{k} \left(\begin{cases}X(g_i^{-1}xg_\alpha), &xg_\alpha \in g_iH \\ 0, & xg_\alpha \not\in g_iH \end{cases} \right) \left( \begin{cases} X(g_\alpha^{-1}yg_j), &y g_j \in g_\alpha H \\ 0, & yg_j \not\in g_\alpha H \end{cases} \right) 
		\end{align*}
		Since the left cosets of $H$ are disjoint, there is exactly one $\alpha$ for which $yg_j \in g_\alpha H$. For that $\alpha$, there is exactly one $i$ such that $xg_\alpha \in g_i H$. In this case, if $x g_\alpha \in g_i H$ and $y g_j \in g_\alpha H$, then it follows that $xy g_j \in g_i H$. Hence:
		\begin{align*}
			\bigl( X\uair_H^G(x) X\uair_H^G(y) \bigr)_{ij}
			&=  
			\begin{cases} 
				X(g^{-1}x g_\alpha) X(g_\alpha ^{-1}y  g_j), & xy g_j \in g_i H \\
				0, & xy g_j \not\in g_i H
			\end{cases}\\
			&=
			\begin{cases}
				X(g_i^{-1}xyg_j), &xyg_j \in g_i H \\
				0, & xyg_j \not\in g_i H
			\end{cases}\\
			&= \bigl( X\uair_H^G(xy) \bigr)_{ij}.
		\end{align*}
		Thus $X\uair_H^G$ is a representation.
	\end{proof}

\begin{exercise}
	This exercise shows that many of our results do not hold over arbitrary fields. Let $X$ be the representation of $\bfZ_3 = \{0,1,2\}$ generated by:
	\[
		X(1) = \begin{bmatrix} -1 & 1 \\ -1 & 0 \end{bmatrix}. 
	\] 
	Show that $X$ is reducible over $\bfC$ but not over $\bfR$.
\end{exercise}
	\begin{proof}
		Consider $X:\bfZ_3 \rightarrow \GL_2(\bfC)$ as above. Recall that a subspace $V \subseteq \bfC^2$ is \emph{$X$-invariant} if $X(g)v \in V$ for all $v \in V$ and all $g \in \bfZ_3$. Since $\lambda = \frac{-1 \pm \sqrt{3}}{2}$ are eigenvalues of $X(1)$ and $X(2)$, $E_\lambda$ is an invariant subspace of $X(1)$ and $X(2)$. Similarly, $E_\lambda$ is an invariant subspace of $X(0)$ because every subspace of $\bfC^n$ is invariant under the identity. Thus $E_\lambda$ is $X$-invariant, hence $X$ is reducible.

		Before considering $X:\bfZ_3 \rightarrow \GL_2(\bfR)$, note that if $X$ is a degree 2 representation, then $X$ is reducible if and only if $X$ is diagonalizable. Indeed, if $X$ were reducible, then $X$ is similar to $A \oplus B$, where $A$ and $B$ are $1\times 1$ block entries of $A \oplus B$. Suppose towards contradiction $X:\bfZ_3 \rightarrow \GL_2(\bfR)$ is reducible. Then $X$ must be diagonalizable. But this is a contradiction, as its characteristic polynomial $x^2 + x + 1$ does not split over $\bfR$. Hence $X$ is irreducible over $\bfR$.
	\end{proof}

\begin{exercise}
	This exercise shows that many of our results do not hold for infinite groups.
	\mbox{}
	\begin{enumerate}[label = (\alph*),itemsep=1pt,topsep=3pt]
		\item Let $G$ be the infinite group $\bfR^+$ with the operation of multiplication. Show that the function $X$ defined by:
			\[
				X(x) := \begin{bmatrix} 1 & 0 \\ \ln x & 1 \end{bmatrix} 
			\] 
			for all $x \in \bfR^+$ is an irreducible representation.
		\item Show that:
			\[
				\Span \left\{ \begin{bmatrix} 0 \\ 1 \end{bmatrix}  \right\}  
			\] 
			is an $X$-invariant subspace of $\bfC^2$, showing Theorem~\ref{} does not hold. Where in the proof of Theorem~\ref{} do we use the finiteness of $G?$
	\end{enumerate}
\end{exercise}
	\begin{proof}
		(a) Suppose towards contradiction $X$ is reducible. For any $\alpha \in \bfR^+$ the characteristic polynomial of $X(\alpha)$ is $(\lambda-1)^2$. Hence $\lambda = 1$ is an eigenvalue. It follows that $E_1 = \Span \left\{ \begin{bmatrix} 0 \\ 1 \end{bmatrix}  \right\} $. Since $\bfC^2$ doesn't have a basis of eigenvectors, $X(\alpha)$ is not diagonalizable for any $\alpha \in \bfR^+$. This contradicts are assumption that $X$ is reducible, since if it were reducible it can be written as a diagonal matrix.

		(b) For any $\alpha \in \bfR^+$, we have $X(\alpha)\begin{bmatrix} 0 \\ 1 \end{bmatrix}  = \begin{bmatrix} 0 \\ 1 \end{bmatrix} $. Hence $\Span \left\{ \begin{bmatrix} 0 \\ 1 \end{bmatrix}  \right\} $ is an $X$-invariant subspace whilst $X$ is irreducible. This shows that Theorem~\ref{} does not hold. In the proof, it may not be the case that $M = \sum_{g \in G}^{} X(g)^\ast X(g)$ and $v^\ast M v$ converges, as $G$ is an infinite group.
	\end{proof}
