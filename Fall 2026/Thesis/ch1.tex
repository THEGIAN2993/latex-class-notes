\section{Basic Definitions and Results}

\begin{definition}
	An \emph{algebra} over $\bbF$ is a vector space $\scrA$ over $\bbF$ that has a multiplication operation making $\scrA$ into a ring such that, for any $\alpha \in \bbF$ and $a,b \in \scrA$, we have $\alpha(ab) = (\alpha a)b = a (\alpha b)$. If $\scrA$ admits an identity element, we say $\scrA$ is \emph{unital}.
\end{definition}

\begin{definition}
	A \emph{normed algebra} is an algebra $\scrA$ over $\bbF$ which has a norm $\lnorm \cdot \rnorm$ making $\scrA$ into a normed linear space and also admitting the property that, for any $a,b \in \scrA$, we have $\lnorm ab \rnorm \leq \lnorm a \rnorm \lnorm b \rnorm$. If $\scrA$ is a Banach space, then $\scrA$ is called a \emph{Banach algebra}. If $\scrA$ has an identity $e$, then it is assumed that $\lnorm e \rnorm = 1$.
\end{definition}

Parts (1) and (2) show that $\scrA$ may be identified isometrically with the ideal $\scrA \times \{0\}$ of the unital algebra $\scrA \times \bbF$, whose identity is $(0,1)$. \section{Ideals and Quotients}

\begin{definition}
	Let $\scrA$ be an algebra. A \emph{left ideal} (resp. \emph{right ideal}) of $\scrA$ is a subalgebra $\scrM$ of $\scrA$ such that, for all $a \in \scrA$ and $x \in \scrM$, then $ax \in \scrM$ (resp. $xa \in \scrM$). If $\scrM$ is both a left and right ideal, then $\scrM$ is simply called an \emph{ideal} (or \emph{bilaterial ideal}).
\end{definition}

\begin{definition}
	Let $\scrA$ be a unital algebra. An element $a \in \scrA$ is \emph{left invertible} (resp. \emph{right invertible}) if there exists an element $x \in \scrA$ such that $xa = 1$ (resp. $ax = 1$). If an element $a \in \scrA$ is both left and right invertible, then $a$ is simply called \emph{invertible}.
\end{definition}

\begin{proposition}\label{prop:ideals-and-inverts}
	Let $A$ be a unital algebra.
	\mbox{}
	\begin{enumerate}[label = (\arabic*),itemsep=1pt,topsep=3pt]
		\item If $a \in \scrA$ is invertible, there is a unique element $a^{-1}$ such that $a a^{-1} = a^{-1} a = 1$.
		\item If $a \in \scrA$ is left invertible, and $\scrM$ is a left ideal in $\scrA$ containing $a$, then $\scrM = \scrA$.
	\end{enumerate}
\end{proposition}
	\begin{proof}
		(1) Since $a$ is invertible, there exists $x,y \in \scrA$ such that $xa = 1 = ay$. Hence $y=1y=(xa)y = x(ay) = x1 = x$.

		(2) Since $a \in \scrA$ is left invertible, there exists $x \in \scrA$ such that $xa = 1$. Since $\scrM$ is a left ideal and $a \in \scrM$, it must be the case that $1 = xa \in \scrM$. Since $1 \in \scrM$, we have $y = y1 \in \scrM$ for all $y \in \scrA$. Thus $\scrA \subseteq \scrM$. 
	\end{proof}

Part (2) of Proposition~\ref{} says that every element of a proper left ideal is necessarily not left invertible.

\begin{lemma}\label{lemma:norm-le-1}
	If $\scrA$ is a unital Banach algebra and $x \in \scrA$ such that $\lnorm x-1 \rnorm < 1$, then $x$ is invertible.
\end{lemma}
	\begin{proof}
		Define $y:=1-x$. Note that:
		\begin{align*}
			\lnorm y \rnorm
			&= \lnorm 1-x \rnorm \\
			&= \lnorm x-1 \rnorm \\
			&< 1.
		\end{align*}
		Since $\scrA$ is a Banach algebra, it follows via induction that:
		\[
			\lnorm y^n \rnorm \leq \lnorm y \rnorm^n.
		\]
		Since $\lnorm y \rnorm < 1$, the series $\sum_{n=0}^{\infty} \lnorm y \rnorm^n$ converges, hence by the comparison test $\sum_{n=0}^{\infty} \lnorm y^n \rnorm$ converges. Because $\scrA$ is complete, $z:=\sum_{n=0}^{\infty}y^n$ also converges. Taking $z_k := \sum_{n=0}^{k}y^n$, we can see:
		\begin{align*}
			z_k(1-y)
			&= \sum_{n=0}^{k} y^n - \sum_{n=0}^{k} y^{n+1} \\
			&= 1-y^{k+1}. \\
		\end{align*}
		It is clear that $\sum_{n=0}^{\infty} y^{n+1}$ converges by the same reasoning from above. Hence $y^{n+1} \rightarrow 0$ as $n \rightarrow \infty$. This gives:
		\begin{align*}
			z(1-y)
			&= \lim_{n \rightarrow \infty}z_n(1-y) \\
			&= \lim_{n \rightarrow \infty}1-y^{n+1} \\
			&= 1. \\
		\end{align*}
		An identical argument will show that $(1-y)z = 1$. Thus $(1-y) = 1-(1-x) = x$ is invertible.
	\end{proof}

The following fact surfaced in the preceeding proof.

\begin{corollary}\label{cor:power-series-basic}
	Let $\scrA$ be a unital Banach algebra. If $a \in \scrA$ and $\lnorm a \rnorm < 1$, then $(1-a)^{-1} = \sum_{k=0}^{\infty} a^k$.
\end{corollary}

\begin{lemma}\label{lemma:inverse-pertubation}
	Let $\scrA$ be a unital Banach algebra. If $b_0 a_0 = 1$ and $\lnorm a - a_0 \rnorm < \lnorm b_0 \rnorm^{-1}$, then $a$ is left invertible. 
\end{lemma}	
	\begin{proof}
		Rearranging $\lnorm a-a_0 \rnorm < \lnorm b_0 \rnorm^{-1}$ yields $\lnorm b_0 a - 1 \rnorm < 1$. By Lemma~\ref{lemma:norm-le-1}, the element $b_0a$ is invertible. Hence there exists $x \in \scrA$ such that $xb_0a = 1$, implying that $a$ is left invertible.
	\end{proof}

\begin{theorem}
	If $\scrA$ is a unital Banach algebra, then:
	\mbox{}
	\begin{enumerate}[label = (\roman*),itemsep=1pt,topsep=3pt]
		\item $G_l := \{a \in \scrA : a \text{ is left invertible }\}$,
		\item $G_r := \{a \in A: a \text{ is right invertible }\}$,
		\item $G := \{a \in \scrA : a \text{ is invertible}\}$,
	\end{enumerate}
	are all open subsets of $\scrA$. Moreover, the map $a \mapsto a ^{-1}$ of $G \rightarrow G$ is continuous.
\end{theorem}
	\begin{proof}
		Let $a_0 \in G_l$. If $a \in B(a_0, \lnorm b_0 \rnorm^{-1})$, then $\lnorm a-a_0 \rnorm < \lnorm b_0 \rnorm^{-1}$, so by Lemma~\ref{lemma:inverse-pertubation} $a$ is left invertible. Thus $a_0 \in B(a_0,\lnorm b_0 \rnorm^{-1}) \subseteq G_l$, establishing $G_l$ as open. It follows similarly that $G_r$ is open, and since $G = G_l \cap G_r$, it is open as well. 

		To prove that $a \mapsto a^{-1}$ is continuous, we first prove an intermediary result. Let $\{a_n\}$ be a sequence in $G$ such that $a_n \rightarrow 1$. Let $0 < \delta <1$ and suppose $\lnorm a_n - 1 \rnorm < \delta$. By Corollary~\ref{cor:power-series-basic}:
		\begin{align*}
			a_n^{-1}
			&= (1-(1-a_n))^{-1} \\
			&= \sum_{k=0}^{\infty} (1-a_n)^k \\
			&= 1+ \sum_{k=1}^{\infty} (1-a_n)^k.
		\end{align*}
		Hence $a_n^{-1} - 1 = \sum_{k=1}^{\infty} (1-a_n)^k$. Thus:
		\begin{align*}
			\lnorm a_n^{-1} - 1 \rnorm
			&= \lnorm \sum_{k=1}^{\infty} (1-a_n)k \rnorm \\
			&\leq \sum_{k=1}^{\infty} \lnorm 1-a_n \rnorm^k \\
			&= \sum_{k=1}^{\infty} \delta^k \\
			&= \frac{\delta}{1-\delta}. 
		\end{align*}
		If $\epsilon > 0$ is given, then $\delta$ can be chosen such that $\frac{\delta}{1-\delta} < \epsilon$. So $\lnorm a_n - 1 \rnorm < \delta$ implies $\lnorm a_n^{-1} - 1 \rnorm < \epsilon$. Hence $a_n^{-1} \rightarrow 1$.
		With this, let $a \in G$ be arbitrary and suppose $\{a_n\}$ is a sequence in $G$ such that $a_n \rightarrow a$. Then $a^{-1}a_n \rightarrow 1$. The intermediary result above gives $a_n^{-1}a = (a^{-1}a_n)^{-1} \rightarrow 1$. Hence $a_n^{-1} = a_n^{-1}aa^{-1} \rightarrow a^{-1}$.
	\end{proof}

\begin{definition}
	A \emph{maximal ideal} is a proper ideal that is contained in no larger proper ideal. The set of maximal ideals of a given algebra (or ring) is denoted by $\Spec_m$.
\end{definition}

\begin{corollary}
	Let $\scrA$ be a unital Banach algebra.
	\mbox{}
	\begin{enumerate}[label = (\arabic*),itemsep=1pt,topsep=3pt]
		\item The closure of a proper left (resp. right, bilateral) ideal is a proper left (resp. right, bilateral) ideal.
		\item A maximal left (resp. right, bilateral) ideal is closed.
	\end{enumerate}
\end{corollary}
	\begin{proof}
		(1) Let $\scrM$ be a proper left ideal of $\scrA$. The proof showing $\cl(\scrM)$ is a subalgebra is omitted, but the idea can be replicated from the following. Let $a \in \scrA$ and $x \in \cl(\scrM)$. There exists a sequence $\{x_n\}$ in $\scrM$ such that $x_n \rightarrow x$. Then $\{a x_n\}$ is a sequence in $\scrM$ such that $a x_n \rightarrow ax$. Thus $ax \in \cl(\scrM)$, so $\cl(\scrM)$ is a left ideal. Let $G_l$ denote the elements of $\scrA$ which are left invertible. By part (2) of Proposition~\ref{prop:ideals-and-inverts}, $\scrM \cap G_l = \emptyset$, which implies that $\scrM \subseteq \scrA \setminus G_l$. Since $\scrA \setminus G_l$ is closed, and since $\cl(\scrM)$ is the smallest closed set containing $\scrM$, we have $\cl(\scrM) \subseteq \scrA \setminus G_l$. Since $G_l$ is necessarily nonempty, $\cl(\scrM) \neq \scrA$, hence it is proper.

		(2) If $\scrM$ is a maximal left ideal, then $\cl(\scrM)$ is a proper ideal by (1). Hence $\scrM = \cl(\scrM)$ by maximality.
	\end{proof}

\begin{theorem}\label{thm:qib}
	If $\scrA$ is a Banach algebra and $\scrM$ is a proper closed ideal in $\scrA$, then $\scrA/\scrM$ is a Banach algebra with norm given by:
	\[
		\lnorm a + \scrM \rnorm := \inf \{\lnorm a + x \rnorm : x \in \scrM\}.
	\] 
	If $\scrA$ has an identity, so does $\scrA/\scrM$.
\end{theorem}
	\begin{proof}
		The proof that $\scrA/\scrM$ is a normed linear space with well-defined operations given by $(a+\scrM) + (b+\scrM) = (a+b) + \scrM$ and $\alpha(a+\scrM) = (\alpha a) + \scrM$ is omitted. We will first show that $\scrA/\scrM$ is a Banach space. Let $\{a_n + \scrM\}_n$ be a Cauchy sequence in $\scrA/\scrM$. There is a subsequence $\{a_{n_k} + \scrM\}_k$ such that:\footnote{Suppose $\{x_n\}$ is a Cauchy sequence. For $\epsilon = 2^{-1}$, choose $N_1$ such that $m,n \geq N_1$ implies $\lnorm x_m - x_n \rnorm < 2^{-1}$. Now choose $n_1 \geq N_1$. For $\epsilon = 2^{-2}$, choose $N_2$ such that $m,n \geq N_2$ implies $\lnorm x_m - x_n \rnorm < 2^{-2}$. Choose $n_2 \geq \max \{N_2, n_1 + 1\}$. Since $n_1, n_2 \geq N_1$, we have $\lnorm x_{n_2} - x_{n_1} \rnorm < 2^{-1}$. Inductively, we obtain a subsequence $\{x_{n_k}\}_k$ satisfying the above propety. }
		\[
			\lnorm (a_{n_k} + \scrM) - (a_{n_{k+1}} + \scrM)\rnorm = \lnorm (a_{n_k} - a_{n_{k+1}}) + \scrM \rnorm < 2^{-k}.	
		\] 
		Let $y_1 = 0$. By the definition of $\lnorm \cdot \rnorm$ on $\scrA/\scrM$, choose $y_2 \in \scrM$ such that:
		\[
			\lnorm a_{n_1} - a_{n_2} + y_2 \rnorm \leq \lnorm a_{n_1} - a_{n_2} + \scrM \rnorm + 2^{-1} < 2 \cdot 2^{-1}.
		\] 
		With $y_2$ fixed, choose $y_3 \in \scrM$ such that:
		\begin{align*}
			\lnorm (a_{n_2} + y_2) - (a_{n_3} - y_2) \rnorm
			&= \lnorm a_{n_2} - a_{n_3} + (y_2 - y_3) \rnorm \\
			&\leq \lnorm a_{n_2} - a_{n_3} + \scrM \rnorm + 2^{-2}  \\
			&< 2 \cdot 2^{-2}.
		\end{align*}
		Inductively, we obtain a sequence $\{y_k\}_k$ in $\scrM$ such that:
		\[
			\lnorm (a_{n_k} + y_k) - (a_{n_{k+1}} + y_{k+1}) \rnorm < 2 \cdot 2^{-k}.
		\] 
		In particular, for $j < k$ we have:
		\begin{align*}
			\lnorm (a_{n_k} + y_j) - (a_{n_k} + y_k) \rnorm
			&\leq \lnorm (a_{n_j} + y_j)-(a_{n_{j+1}} + y_{j+1}) \rnorm + \ldots +\lnorm (a_{n_{k-1}} + y_{k-1}) - (a_{n_k} + y_k) \rnorm  \\
			&\leq \sum_{i=j}^{k} 2\cdot_2^{-k}  \\
			&\leq \sum_{i=j}^{\infty} 2\cdot_2^{-k}  \\
			&= 2^{2-j}.
		\end{align*}
		Thus $\{a_{n_k} + y_k\}_k$ is a Cauchy sequence in $\scrA$. Since $\scrA$ is complete, there exists $a_0 \in \scrA$ such that $a_{n_k} + y_k \rightarrow a_0$. Since the projection $\pi:\scrA \rightarrow \scrA/\scrM$ is continuous, we have:
		\begin{align*}
			\lim_{k \rightarrow \infty}a_{n_k} + \scrM
			&= \lim_{k \rightarrow \infty} \pi(a_{n_k} + y_k) \\
			&= \pi\left(\lim_{k \rightarrow \infty} a_{n_k} + y_k \right) \\
			&= \pi(a_0) \\
			&= a_0 + \scrM.
		\end{align*}
		Since $\{a_n + \scrM\}_n$ is a Cauchy sequence with a convergent subsequence, it is convergent. Thus $\scrA/\scrM$ is a Banach space.

		The proof that $\scrA/\scrM$ is an algebra with well-defined multiplication given by $(a+\scrM)(b+\scrM) = ab + \scrM$ is omitted. Let $a,b \in \scrA$ and $u,v \in \scrM$. Then $(a+u)(b+v) = ab + (av + ub + uv) \in ab+\scrM$. Hence:
		\begin{align*}
			\lnorm (a+\scrM)(b+\scrM) \rnorm
			&= \lnorm ab + \scrM \rnorm \\
			&\leq \lnorm (a+u)(b+v) \rnorm.
		\end{align*}
		Taking the infimum over all $u,v \in \scrM$ gives $\lnorm (a+\scrM)(b+\scrM) \rnorm \leq \lnorm a+\scrM \rnorm \lnorm b+\scrM \rnorm$. Thus $\scrA/\scrM$ is a Banach algebra. Moreover, $1 + \scrM$ is indeed the identity of $\scrA/\scrM$ as $(1+\scrM)(a+\scrM) = (1a + \scrM) = a + \scrM = (a1 + \scrM) = (a + \scrM)(1 + \scrM)$
	\end{proof}

\section{The Spectrum}

\begin{definition}
	If $\scrA$ is a unital Banach algebra and $a \in \scrA$, the \emph{spectrum of $a$} is:
	\[
		\sigma(a) := \{a \in \bbF : a - \alpha \text{ is not invertible }\}.
	\] 
	The \emph{left spectrum of $a$} is:
	\[
		\sigma_l(a) := \{a \in \bbF : a - \alpha \text{ is not left invertible }\}.
	\] 
	The \emph{right spectrum of $a$}, denoted $\sigma_r(a)$, is defined similarly.
\end{definition}

\begin{definition}
	If $\scrA$ is a unital Banach algebra and $a \in \scrA$, the \emph{resolvent set of $a$} is $\rho(a) := \bbF \setminus \sigma(a)$. The \emph{left} and \emph{right resolvents of $a$} are $\rho_l(a) := \bbF \setminus \sigma_l(a)$ and $\rho_r(a) := \bbF \setminus \sigma_r(a)$.
\end{definition}

\begin{example}
	Let $X$ be compact. If $f \in C(X)$, then $\sigma(f) = f(X)$. Indeed, let $\alpha \in f(X)$. Then $\alpha = f(x_0)$ for some $x_0 \in X$. Hence $0 = f(x_0) - \alpha \in (f-\alpha)(X)$. It must be the case that $(f-\alpha)^{-1}$ does not exist, so $\alpha \in \sigma(f)$. Conversely, if $f-\alpha$ is not invertible, then $(f-\alpha)^{-1}$ does not exist. This means $0 = f(x_0) - \alpha$ for some $x_0 \in X$, hence $\alpha = f(x_0)$. 
\end{example}

\begin{example}
	Let $X$ be a Banach space. If $A \in \scrB(X)$, then $\sigma(A) = \{\alpha \in \bbF : \text{ either }\ker(A-\alpha) \neq 0\text{ or }\Image(A-\alpha) \neq X\}$. Indeed, if $A - \alpha$ is not invertible, then it is either not injective or not surjective. In fact, this means that $\rho(A) = \bbF \setminus \sigma(A) = \{\alpha \in \bbF : A - \alpha \text{ is bijective }\}$. If $\alpha \in \rho(A)$, then $\alpha \not\in \sigma(A)$, hence $\ker(A - \alpha) = 0$ and $\Image(A - \alpha) = X$. Hence $A - \alpha$ is bijective. Conversely, suppose $A - \alpha$ is bijective. By the Open Mapping Theorem, $A-\alpha$ is an open map. For any open subset $U \subseteq X$, the set $((A-\alpha)^{-1})^{-1}(U) = (A-\alpha)(U)$ is open. Thus $(A-\alpha)^{-1} \in \scrB(X)$, hence $\alpha \in \rho(A)$.
\end{example}

\section{The Riesz Functional Calculus}

\begin{definition}
	Let $G \subseteq \bbC$ be open. A closed rectifiable curve $\gamma$ is \emph{positively oriented} if for every $a \in G \setminus \{\gamma\}$, $n(\gamma;a)$ is either $0$ or $1$. In this case the \emph{inside} of $\gamma$ is $\ins \gamma := \{a \in \bbC \setminus \{\gamma\} : n(\gamma;a) = 1\}$. The \emph{outside} of $\gamma$ is $\out \gamma := \{a \in \bbC \setminus \{\gamma\}: n(\gamma;a) = 0\}$.
\end{definition}

\begin{definition}
	A curve $\gamma:[0,1] \rightarrow \bbC$ is \emph{simple} if $\gamma(s) = \gamma(t)$ implies either:
	\mbox{}
	\begin{itemize}[itemsep=1pt,topsep=3pt]
		\item $s = t$,
		\item $s=0$ and $t = 1$, or
		\item $s=1$ and $t = 0$.
	\end{itemize}
\end{definition}

\begin{definition}
	If $\Gamma = \{\gamma_1,\ldots ,\gamma_m\}$ is a collection of closed rectifiable curves, then $\Gamma$ is \emph{positively oriented} if:
	\mbox{}
	\begin{itemize}[itemsep=1pt,topsep=3pt]
		\item $\{\gamma_i\}\cap \{\gamma_j\} = \emptyset$,
		\item for all $a \in \bbC \setminus  \bigcup_{j=1}^m \{\gamma_j\}$, $n(\Gamma;a) := \sum_{j=1}^{m} n(\gamma_j;a)$ is either $0$ or $1$, and
		\item each $\gamma_j$ is a simple curve.
	\end{itemize}
	The \emph{inside} of $\Gamma$ is $\ins \Gamma := \left\{a \in \bbC \setminus \bigcup_{j=1}^m \{\gamma)j\right\}: n(\Gamma;a) = 1\}$. The \emph{outside} of $\Gamma$ is $\out \Gamma := \left\{a \in \bbC \setminus \bigcup_{j = 1}^m \{\gamma_j\} : n(\Gamma;a) = 0\right\}$. 
\end{definition}

The curves $\gamma_1,\ldots ,\gamma_m$ can be found such that they are infinitely differentiable.
\end{proposition}

\begin{definition}
	Let $\scrA$ be a unital Banach algebra and let $a \in \scrA$. The set of functions that are analytic in a neighborhood of $\sigma(a)$ is denoted by $\Hol(a)$.
\end{definition}

\begin{remark}
	$\Hol(a)$ is not a Banach algebra. The domain of $f+g$ and $fg$ is $\Dom(f) \cap \Dom(g)$, where $\Dom(f)$ and $\Dom(g)$ are the domains of $f$ and $g$, respectively.
\end{remark}

\begin{theorem}[Riesz Functional Calculus]
	Let $\scrA$ be a unital Banach algebra and $a \in \scrA$. Define $\rho:\Hol(a) \rightarrow \scrA$ by:
	\[
		f \mapsto \frac{1}{2 \pi i}\int_{\Gamma}^{} f(z)(z-a)^{-1}\,dz,
	\] 
	where, using Proposition~\ref{prop:existence-of-posoc}, $\Gamma$ is a positively oriented collection of curves in an open set $G \subseteq \bbC$ such that $\sigma(a) \subseteq \ins \Gamma \subseteq G$.
	\mbox{}
	\begin{enumerate}[label = (\alph*),itemsep=1pt,topsep=3pt]
		\item The map $\rho$ is well-defined; i.e., if $G \subseteq \bbC$ is open such that $\sigma(a) \subseteq G$, $\Gamma = \{\gamma_1,\ldots ,\gamma_m\}$ and $\Lambda = \{\lambda_1,\ldots ,\lambda_k\}$ are two positively oriented collections of curves in $G$ such that $\sigma(a) \subseteq \ins \Gamma \subseteq G$ and $\sigma(a) \subseteq \ins \Lambda \subseteq G$, then:
			\[
				\frac{1}{2 \pi i}\int_{\Gamma}^{} f(z)(z-a)^{-1}\,dz = \frac{1}{2 \pi i} \int_{\Lambda}^{} f(z)(z-a)^{-1}\,dz. 
			\]

		\item The map $\rho$ is an algebra homomorphism.
		\item If $f(z) = z^k$ for all $z$, then $\rho(f) = a^k$.
		\item If $\{f,f_1,f_2,\ldots \}$ is a sequence of analytic functions on an open set $G$ such that $\sigma(a) \subseteq G$ and $f_n \rightarrow f$ uniformly on compact subsets of $G$, then $\rho(f_n) \rightarrow \rho(f)$.

		\item If $f(z) = \sum_{k=0}^{\infty} \alpha_k z^k$ has radius of convergence strictly greater than $r(a)$, then $f \in \Hol(a)$ and $\rho(f) = \sum_{k=0}^{\infty}\alpha_k a^k$.

\begin{proposition}
	If $a,b \in \scrA$ such that $ab=ba$, and $f \in \Hol(a)$, then $f(a)b = bf(a)$.

Then $q(a)b = bq(a)$ as was just shown, but by mulyiplying $\bigl(q(a)\bigr)^{-1}$ on both sides we see that $\bigl( q(a) \bigr)^{-1} b = b \bigl( q(a) \bigr)^{-1}$. 		\begin{align*}
			r(a)b
			&= (pq^{-1})(a) b \\
			&= p(a)q^{-1}(a) b\\
			&= bp(a)q^{-1}(a) \\
			&= br(a).
		\end{align*}
		Using Runge's Theorem, if $f \in \Hol(a)$ such that $f$ is analytic on an open set $G$ containing $\sigma(a)$, there is a sequence $\{f_n\}$ of rational functions which converge to $f$ on compact subsets of $G$. Since $f_n(a)b = bf_n(a)$ for each $n$, we have:

			f(a)b
			&= \lim_{n \rightarrow \infty}f_n(a)b \\
			&= \lim_{n \rightarrow \infty}b f_n(a) \\
			&= bf(a) \qedhere
		\end{align*}i

\begin{theorem}[The Spectral Mapping Theorem]
	If $a \in \scrA$ and $f \in \Hol(a)$, then $\sigma(f(a)) = f(\sigma(a))$.
\end{theorem}
	\begin{proof}
		Let $f(\alpha) \in \sigma(f(a))$. Then $\alpha \in \sigma(a)$. Towards contradiction, suppose $f(\alpha) \not\in \sigma(f(a))$. Then $f(a) - f(\alpha)$ is invertible for some $z \in \bbC$. Consider a function $g$ defined by $f(z) - f(\alpha) = (z-\alpha)g(z)$. Then $g$ is analytic on a neighborhood of $\sigma(a)$; i.e., $g \in \Hol(a)$. The Riesz Functional Calculus of $g$ is $f(a) - f(\alpha) = (a-\alpha)g(a)$. Hence $(a-\alpha)$ is invertible with inverse $g(a) \bigl( f(a) - f(\alpha) \bigr)^{-1}$. This contradicts $\alpha \in \sigma(a)$, concluding that $f(\alpha) \in \sigma(f(a))$. Thus $f(\sigma(a)) \subseteq \sigma(f(a))$. 

		Conversely, suppose $\beta \not\in f(\sigma(a))$. Then $\beta \neq f(z)$ for every $z \in \sigma(a)$. Hence $f(z) - \beta$ is nonzero on $\sigma(a)$. Thus $g(z) := \bigl( f(z) - \beta \bigr)^{-1}$ is in $\Hol(a)$. After applying the Riesz Functional Calculus, we get $g(a)\bigl( f(a) - \beta \bigr)^{-1} = 1$. Hence $\beta \not\in \sigma(f(a))$, establishing $\sigma(f(a)) \subseteq f(\sigma(a))$.
	\end{proof}

\section{Abelian Banach Algebras}

\noindent Throughout, assume that all Banach algebras contain an identity.

\begin{theorem}[Gelfand-Mazur]\label{thm:gm}
	If $\scrA$ is a Banach algebra that is also a division ring, then $\scrA$ is isometrically isomorphic to $\bbC$. 
\end{theorem}
	\begin{proof}
		Define $\varphi:\bbC \rightarrow \scrA$ by $z \mapsto z 1_{\scrA}$. This is clearly an algebra homomorphism, and moreover:
		\begin{align*}
			\lnorm \varphi(z) \rnorm
			&= |z| \lnorm 1_\scrA \rnorm \\
			&= |z|.
		\end{align*}
		Hence $\varphi$ is an isometry (and thus an injection). It remains to show surjectivity. If $a \in \scrA$, then $\sigma(a) \neq 0$. If $z \in \sigma(a)$, then $a-z$ is not invertible. But $\scrA$ is a division ring; i.e., $\scrA^\times = \scrA \setminus \{0\}$. Thus $a-z = 0$, hence $a = z = z 1_\scrA = \varphi(z)$.
	\end{proof}

\begin{corollary}\label{cor:max-iff-c}
	Let $\scrA$ be an abelian Banach algebra and $\scrM$ a proper ideal. Then $\scrM$ is maximal if and only if $\scrA/\scrM$ is isometrically isomorphic to $\bbC$.
\end{corollary}
	\begin{proof}
		The ideal $\scrM$ is maximal if and only if there are no ideals $I$ with $\scrM \subset I \subset \scrA$. The ideals of $\scrA$ containing $\scrM$ correspond bijectively with the ideals of $\scrA/\scrM$, so $\scrM$ is maximal if and only if the ideals of $\scrA/\scrM$ are $0$ and $\scrA/\scrM$. Equivalently, $\scrM$ is maximal if and only if $\scrA/\scrM$ is a field. By Theorem~\ref{thm:qib} and the \nameref{thm:gm} Theorem, $\scrM$ is maximal if and only if $\scrA/\scrM \cong \bbC$.
	\end{proof}

\begin{definition}
	A \emph{character} of an abelian Banach algebra $\scrA$ is a nonzero homomorphism $\chi:\scrA \rightarrow \bbC$. The set of all characters of $\scrA$ is denoted by $\Sigma_\scrA$.
\end{definition}

\begin{lemma}\label{lem:character-surj}
	Every $\chi \in \Sigma_\scrA$ is surjective.
\end{lemma}
	\begin{proof}
		Let $z \in \bbC$. For any $a \in A$ we have $\chi \bigl( \frac{z a}{\chi(a)} \bigr) = z \frac{\chi(a)}{\chi(a)} = z$.
	\end{proof}

\begin{proposition}\label{prop:character-maximal}
	Let $\scrA$ be an abelian Banach algebra and $\scrM$ a maximal ideal of $\scrA$. Then $\scrM = \ker \chi$ for some $\chi \in \Sigma_\scrA$.
\end{proposition}
	\begin{proof}
		Let $\pi:\scrA \rightarrow \scrA/\scrM$ be the natural projection. By Corollary~\ref{cor:max-iff-c} and the \nameref{thm:gm} Theorem, $\varphi:\scrA/\scrM \rightarrow \bbC$ given by $z 1_\scrA + \scrM \mapsto z$ is an isomorphism. Define $\chi:\scrA \rightarrow \bbC$ by $\chi := \varphi \circ \pi$. Then $\chi \in \Sigma_\scrA$ and:
		\begin{align*}
			\ker \chi
			&= \{a \in A : \chi(a) = 0\} \\
			&= \{a \in A : \varphi(a + \scrM) = 0\} \\
			&= \scrM. \qedhere
		\end{align*}
	\end{proof}

\begin{proposition}
	If $\chi \in \Sigma_\scrA$, then $\ker \chi$ is a maximal ideal.
\end{proposition}
	\begin{proof}
		By the First Isomorphism Theorem, $\scrA/\ker \chi \cong \chi(\scrA)$. Lemma~\ref{lem:character-surj} says that $\chi(\scrA) = \bbC$, hence by Corollary~\ref{cor:max-iff-c} $\ker \chi$ is maximal.
	\end{proof}

\begin{theorem}
	Let $\scrA$ be an abelian Banach algebra. There is a bijection between $\Sigma_\scrA$ and $\Spec_m(\scrA)$.
\end{theorem}
	\begin{proof}
		Define $f:\Sigma_\scrA \rightarrow \Spec_m(\scrA)$ by $\chi \mapsto \ker \chi$. Suppose $\chi_1,\chi_2 \in \Sigma_A$ with $f(\chi_1) = f (\chi_2)$. Then $\ker \chi_1 = \ker \chi_2$. Note that for any $a \in \scrA$, if $\chi(a) = 0$ then $a \in \ker \chi_1 = \ker \chi_2$, hence $\chi_2(a) = 0$. By Corollary~\ref{cor:lin-funct-rel}, $\chi_1 = \alpha \chi_2$. But notice $\alpha = \alpha \chi_2(1) = \chi_1(1) = 1$, hence $\chi_1 = \chi_2$. Thus $f$ is injective. Proposition~\ref{prop:character-maximal} shows that $f$ is surjective, hence $\Sigma_\scrA$ and $\Spec_m(\scrA)$ have a bijective correspondence.
	\end{proof}

\begin{corollary}
	If $\scrA$ is an abelian Banach algebra, then $\Sigma_\scrA \subseteq \scrA^\ast$; i.e., every character of $\scrA$ is continuous.
\end{corollary}
	\begin{proof}
		Let $\chi \in \Sigma_\scrA$. Then $\ker \chi$ is a maximal ideal, hence it is closed. If it were the case that $\chi$ is not continuous, then it is unbounded. Hence there exists a sequence $\{a_n\}_n \subseteq \scrA$ such that $\chi(a_n) \rightarrow \infty$ and $n \rightarrow \infty$. If $a \not\in \ker \chi$, consider the sequence $\{a'_n\}_n$ given by:
		\[
			a'_n := a- \frac{\chi(a)}{\chi(a_n)}a_n.
		\] 
		Then $\chi(a_n') = 0$ for all $n$, hence $\{a_n'\}_n \subseteq \ker\chi$. But $a'_n \rightarrow a \not\in \ker \chi$, so $\ker \chi$ is not closed. Since we've proven the contrapositive, every $\chi \in \Sigma_\scrA$ is continuous; i.e., $\Sigma_\scrA \subseteq \scrA^\ast$.
	\end{proof}
