\begin{problem}
	Prove Cauchy's inequality with $\epsilon > 0$: If $a,b \in \bfR$ then:
	\[
		ab \leq \frac{\epsilon a^2}{2} + \frac{b^2}{2\epsilon}.
	\] 
\end{problem}
	\begin{proof}
		Since $a,b \in \bfR$, we have:
		\[
			0 \leq (a-b)^2.
		\] 
		Expanding the right-hand side and isolating the $ab$ term yields:
		\[
			ab \leq \frac{a^2}{2} + \frac{b^2}{2}.
		\]
		Now let $\epsilon > 0$. Since this holds true for all $a',b' \in \bfR$, take $a' = \sqrt{\epsilon}a$ and $b' = \frac{b}{\sqrt{\epsilon}}$. Then:
		\begin{align*}
			ab
			&= \sqrt{\epsilon}a \cdot \frac{b}{\sqrt{\epsilon}} \\
			& \leq \frac{(\sqrt{\epsilon}a)^2}{2} + \frac{\left( \frac{b}{\sqrt{\epsilon}} \right)^2}{2} \\
			& = \frac{\epsilon a ^2}{2} + \frac{b^2}{2 \epsilon}. \qedhere
		\end{align*}
	\end{proof}

\begin{problem}
	Let $U \subset \bfR^n$ be open, bounded, and $\partial U$ is $C^1$. Let $u \in C^2(\overline{U})$ with $u = 0$ on $\partial U$. Prove the following interpolation inequality:
	\[
		\int_{U}^{} |Du|^2\,dx \leq \frac{\epsilon}{2} \int_{U}^{} (\Delta u )^2\,dx + \frac{1}{2\epsilon}\int_{U}^{} u^2\,dx
	\] 
\end{problem}
	\begin{proof}
		From the previous problem, take $a = u(x)$ and $b = \Delta u(x)$. Then:
		\[
			-u(x) \Delta u(x) \leq \frac{\epsilon (\Delta u(x))^2}{2} + \frac{(-u(x))^2}{2\epsilon}.
		\]
		Integrating both sides over $U$ gives:
		\[
			- \int_{U}^{} u \Delta u\,dx \leq \frac{\epsilon}{2}\int_{U}^{} (\Delta u)^2\,dx + \frac{1}{2\epsilon} \int_{U}^{} u^2\,dx
		\]
		Since $u = 0$ on $\partial U$, we can rewrite the left-hand side of the inequality as:
		\[
			- \int_{U}^{} u \Delta u\,dx + \int_{\partial U}^{} \frac{\partial u}{\partial \nu}u\,dS \leq \frac{\epsilon}{2}\int_{U}^{} (\Delta u)^2\,dx + \frac{1}{2\epsilon} \int_{U}^{} u^2\,dx.
		\]
		Green's identity yields:
		\[
			\int_{U}^{} Du \cdot Du\,dx \leq \frac{\epsilon}{2}\int_{U}^{} (\Delta u)^2\,dx + \frac{1}{2\epsilon} \int_{U}^{} u^2\,dx.
		\]
		Rewriting $Du \cdot Du = |Du|^2$ gives the desired inequality.
	\end{proof}

\begin{problem}
	Assuming $f$ aproaches zero rapidly enough as $x \rightarrow \pm \infty$, find a function $u$ which satisfies the ODE problem:
	\[
		u''(x) - u(x) = f(x).
	\] 
	Your solution should be of the form:
	\[
		 u(x) = \int_{-\infty}^{\infty} K(x,t)f(x-t)dt.
	\] 
\end{problem}
	\begin{proof}
		After taking the Fourier transform of both sides, our ODE becomes:
		\[
			(iy)^2 \hat{u} - \hat{u} = \hat{f}.
		\]
		Solving for $\hat{u}$ gives:
		\[
			\hat{u}(y) = \frac{-1}{1+y^2}\hat{f}.
		\] 
		Define $h(y) = \frac{-1}{1+y^2}$. Taking the inverse Fourier transform of both sides yields:
		\begin{align*}
			u(x)
			&= \cF^{-1}(h(y)\hat{f}(y))\\
			&= \left( \check{h} \ast f \right)(x) \\
			&= \int_{-\infty}^{\infty} \check{h}(t) f(x-t) dt.
		\end{align*}
		We previously showed that $\cF(e^{-a|x|}) = \frac{1}{\sqrt{2\pi}}\frac{2a}{a^2 + y^2}$. Hence:
		\begin{align*}
			\check{h}(y) = \cF^{-1}\left(\frac{-1}{1 + y^2}\right) = -\frac{\sqrt{2\pi}}{2} e^{-|x|}.
		\end{align*}
		Thus the solution to our ODE is:
		\[
			u(x) = -\frac{\sqrt{2\pi}}{2}\int_{-\infty}^{\infty} e^{-|t|}f(x-t)dt. \qedhere
		\] 
	\end{proof}

\begin{problem}
	Let $a > 0$ and $f \in L^1(\bfR)$. Define $g(x) := f(ax)$. Show that:
	\[
		\hat{g}(y) = \frac{1}{a}\hat{f}\left( \frac{y}{a} \right).
	\] 
\end{problem}
	\begin{proof}
		Observe that:
		\begin{align*}
			\hat{g}(y)
			&= \frac{1}{\sqrt{2\pi}} \int_{-\infty}^{\infty} g(x)e^{-ixy}\,dx \\
			&= \frac{1}{\sqrt{2\pi}} \int_{-\infty}^{\infty} f(ax)e^{-ixy\,}dx.
		\end{align*}
		The substition $x = \frac{t}{a}$ gives $dx = \frac{1}{a}dt$. Hence:
		\begin{align*}
			\hat{g}(y)
			&= \frac{1}{\sqrt{2\pi}} \int_{-\infty}^{\infty} \frac{1}{a}f(t)e^{-it \frac{y}{a}}\,dy \\
			&= \frac{1}{a}\hat{f}\left( \frac{y}{a} \right). \qedhere
		\end{align*}
	\end{proof}

\begin{problem}
	For $x \in \bfR$ and $\epsilon > 0$, let $v_\epsilon(x) := e^{-\epsilon x^2}$. Use the previous two problems to show that:
	\[
		\widehat{v_\epsilon}(y) = \frac{1}{\sqrt{2\epsilon}}e^{\frac{-y^2}{4\epsilon}}.
	\]	
\end{problem}
	\begin{proof}
		Let $f(x) = e^{-\frac{1}{2}x^2}$ Then $v_\epsilon(x) = f\left(\sqrt{2\epsilon}x\right)$. By Problem 4 and 5, we have: 
		\begin{align*}
			\widehat{v_\epsilon}(y) 
			&= \cF\left(f\left(\sqrt{2\epsilon}x\right)\right) \\
			&= \frac{1}{\sqrt{2 \epsilon}} \widehat{f}\left( \frac{y}{\sqrt{2\epsilon}} \right) \\
			&= \frac{1}{\sqrt{2 \epsilon}} f\left( \frac{y}{\sqrt{2\epsilon}} \right) \\
			&= \frac{1}{\sqrt{2\epsilon}}e^{\frac{-y^2}{4 \epsilon}}. \qedhere
		\end{align*}
	\end{proof}

\begin{problem}
	For $x \in \bfR^n$ and $\epsilon > 0$, let $v_\epsilon(x) := e^{- \epsilon|x|^2}$. Use Fubini's theorem and the previous problem to show that:
	\[
		\widehat{v_\epsilon}(y) = \frac{1}{(2\epsilon)^{\frac{n}{2}}}e^{\frac{-|y|^2}{4\epsilon}}.
	\]
\end{problem}
	\begin{proof}
		First note that we can rewrite $v_\epsilon(x)$ and $e^{-i x \cdot y}$ as:
		\begin{align*}
			v_\epsilon(x) &= e^{-\epsilon|x|^2} = \prod_{j=1}^n e^{-x_j^2}, \\
			e^{-i x \cdot y} & = \prod_{j=1}^n e^{-ix_jy_j}.
		\end{align*}
		Using this fact and Fubini's theorem yields:
		\begin{align*}
			\widehat{v_\epsilon}(y)
			&= \frac{1}{(2\pi)^{\frac{n}{2}}}\int_{\bfR^n}^{} v_\epsilon(x)e^{-i x \cdot y}dx \\
			&= \left( \frac{1}{\sqrt{2\pi}} \right)^n \int_{-\infty}^{\infty} \ldots \int_{-\infty}^{\infty} \prod_{j=1}^n	e^{-x_j^2}e^{-i x_j y_j}\,dx_1\ldots dx_n.
		\end{align*}
		For a fixed $j$, note that $\int_{-\infty}^{\infty} e^{x_j^2}e^{-i x_j y_j}dx_j$ is viewed as a constant when integrated against any other $x_k$ such that $k \neq j$. Using this fact, and by Problem 6, we have:
		\begin{align*}
			\widehat{v_\epsilon}
			&=  \left( \frac{1}{\sqrt{2\pi}} \right)^n \int_{-\infty}^{\infty} \ldots \int_{-\infty}^{\infty} \prod_{j=1}^n	e^{-x_j^2}e^{-i x_j y_j}\,dx_1\ldots dx_n\\
			&= \prod_{j=1}^n \frac{1}{\sqrt{2\pi}}\int_{-\infty}^{\infty} e^{-x_j^2}e^{-i x_j y_j}dx_j \\
			&= \prod_{j=1}^n \frac{1}{\sqrt{2\epsilon}}e^{\frac{-y_j^2}{4\epsilon}} \\
			&= \frac{1}{(2\epsilon)^{\frac{n}{2}}}e^{\frac{-|y|^2}{4\epsilon}}. \qedhere
		\end{align*}
	\end{proof}

\begin{problem}
	Show that $\int_{-\infty}^{\infty} e^{-x^2}dx = \sqrt{\pi}$. Then use this to show that:
	\[
		\int_{\bfR^n}^{} e^{-|x|^2}dx = \pi^{\frac{n}{2}}.
	\] 
\end{problem}
	\begin{proof}
		Let $I = \int_{-\infty}^{\infty} e^{-x^2}dx$. Then:
		\begin{align*}
			I^2
			&= \left( \int_{-\infty}^{\infty} e^{-x^2}\,dx \right)^2  \\
			&= \left( \int_{-\infty}^{\infty} e^{-x^2}\,dx \right) \left( \int_{-\infty}^{\infty}e^{-y^2}\,dy  \right)   \\
			&= \int_{-\infty}^{\infty} \int_{-\infty}^{\infty} e^{-(x^2+y^2)}\,dxdy \\
			&= \int_{0}^{2\pi} \int_{0}^{\infty} e^{-r^2}r\,drd\theta \\
			&= \int_{0}^{2\pi} \left( \frac{1}{2}\int_{0}^{\infty} e^{-u}\,du \right) \,d\theta \\
			&= \int_{0}^{2\pi} \frac{1}{2}\,d\theta \\
			&= \pi.
		\end{align*}
		Thus $I = \sqrt{\pi}$. Hence:
		\begin{align*}
			\int_{\bfR^n}^{} e^{-|x|^2}\,dx
			&= \int_{-\infty}^{\infty} \ldots\int_{-\infty}^{\infty} \prod_{j=1}^ne^{-x_j^2}\,dx_1\ldots dx_n \\
			&= \prod_{j=1}^n \int_{-\infty}^{\infty} e^{x_j^2}dx_j \\
			&= \prod_{j=1}^n \sqrt{\pi} \\
			&= \pi^{\frac{n}{2}}. \qedhere
		\end{align*}
	\end{proof}
