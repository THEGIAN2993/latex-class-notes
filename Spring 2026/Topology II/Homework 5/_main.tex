\begin{problem}[manual-num={1}]
	Prove that the fundamental group of the $n$-linked sausage is isomorphic to the free group on $n$ generators.
\end{problem}
	\begin{proof}
		(I am assuming in this proof that the $n$-linked sausages are flat). We proceed by induction on $n$. For the base case, let $n = 2$. 
	\end{proof}
