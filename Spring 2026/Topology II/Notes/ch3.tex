\section{Homotopy of Functions and Paths}

Unless otherwise stated, assume $I:=[0,1]$.

\begin{definition}
    If $f$ and $g$ are continuous maps from topological spaces $X$ to $Y$, we say $f$ is \textui{homotopic} to $g$ if there exists a continuous map $F:X \times I \rightarrow Y$ such that $F(x,0) = f$ and $F(x,1) = g$. We say $F$ is a \textui{homotopy} between $f$ and $g$, and write $f \simeq g$. If $f$ is homotopic to a constant map, we say $f$ is \textui{nullhomotopic}.
\end{definition}

\begin{example}{\color{red}***}
    Let $X$ be a topological space and $Y$ a convex subset of a topological vector space. Then for any two functions $f,g:X \rightarrow Y$, we have $f \simeq g$. Indeed, define $F:X \times I \rightarrow Y$ by:
        \begin{equation*}
        \begin{split}
            F(x,t) := (1-t)f(x) + tg(x).
        \end{split}
        \end{equation*}
    Since addition and scalar multiplication are continuous, $F$ is continuous. Moreover, since $Y$ is convex, $F(x,t) \in Y$ for all $t \in [0,1]$. We call $F$ the \textit{straight line homotopy} between $f$ and $g$.
\end{example}

\begin{example}
    Let $i:S^1 \rightarrow \bfR^2 \setminus \{(0,0)\}$ be the identity map. Define $f:S^1 \rightarrow \bfR^2 \setminus \{(0,0)\}$ by $f(x) = (1,0)$. Then $i \not\simeq f$.
\end{example}

\begin{definition}
    A \textui{path} is a continuous map $f:I \rightarrow X$. We call $f(0)$ the \textui{initial point} and $f(1)$ the \textui{terminal point}.
\end{definition}

\begin{definition}
    Two paths $f,g:I \rightarrow X$ are \textui{path homotopic} if they have the same initial point $x_0$ and same terminal point $x_1$, and there exists a continuous map $F:I \times I \rightarrow X$ such that:
        \begin{equation*}
        \begin{split}
            F(0,t) &= x_0\h9\h7F(1,t) = x_1\\
            F(s,0) &= f(s)\h9\h1F(s,1) = g(s)\\
        \end{split}
        \end{equation*}
    for all $s,t \in [0,1]$. We call $F$ a \textui{path homotopy} and write $f \simeq_p g$.
\end{definition}

\begin{definition}
    If $f$ is a path in $X$ from $x_0$ to $x_1$ and $g$ is a path in $X$ from $x_1$ to $x_2$, we define the \textui{product} of $f$ and $g$ to be the path $f \ast g :I \rightarrow X$ given by:
        \begin{equation*}
        \begin{split}
            (f \ast g)(s) 
            & = \begin{cases}
            f(2s), & 0 \leq s \leq \frac12 \\
            g(2s - 1), & \frac12 \leq s \leq 1
            \end{cases}.
        \end{split}
        \end{equation*}
\end{definition}

\begin{theorem}\label{thm:groupoid}
    The operation $\ast$ is well-define on $C(I,X)/\simeq_p$, defined by the equation $[f] \ast [g] = [f \ast g]$. Moreover, the operation $\ast$ satisfies the following properties:
    \begin{enumerate}[label = (\alph*),itemsep=1pt,topsep=3pt]
        \item If $[f] \ast([g] \ast [h])$ is defined, so if $([f] \ast [g]) \ast [h]$, and they are equal (Associativity).
        \item Given $x_0 \in X$, let $e_{x_0}$ denote the constant path $e_{x_0}:I \rightarrow X$ given by $e_{x_0}(s) = x_0$. If $f$ is a path in $X$ for $x_0$ to $x_1$, then $[f] \ast [e_{x_1}] = [f]$ (Identities).
        \item Given a path $f:I \rightarrow X$ from $x_0$ to $x_1$, define $\overline{f}:I \rightarrow X$ by $\overline{f}(s) = f(1-s)$. Then $[f] \ast [\overline{f}] = [e_{x_0}]$ and $[\overline{f}] \ast [f] = [e_{x_1}]$ (Inverses).
    \end{enumerate}
\end{theorem}

    \end{proof}}

\begin{remark}
    Theorem~\ref{thm:groupoid} establishes that $C(I,X)/\simeq_p$ is a \textit{groupoid}. Here $\ast$ is not a binary operation as it is not necessarily defined for all pairs of elements in $C(I,X)/\simeq_p$. Furthermore, the identity element is not unique for all path-homotopy equivalences.
\end{remark}

\section{The Fundamental Group}

\begin{definition}
    Let $X$ be a topological space. Fix a point $x_0 \in X$. A path in $X$ that begins and ends at $x_0$ is called a \textui{loop in $X$ based at $x_0$}. The set of all path homotopies of loops based at $x_0$ is denoted $\Omega(x_0) := \{\alpha \mid \text{$\alpha$ is a loop in $X$ based at $x_0$ }\}$.
\end{definition}

\begin{theorem}
    The operation $\ast$ is a binary operation on $\Omega(x_0)/\simeq_p$, and and the pair $(\Omega(x_0)/\simeq_p,\ast)$ is a group.
\end{theorem}
    \begin{proof}
        This follows from Theorem~\ref{thm:groupoid}. Clearly $\ast$ is well-defined for any two elements in $\Omega(x_0)/\simeq_p$, and $[e_{x_0}]$ is the unique identity element.
    \end{proof}

\begin{definition}
    The quotient $\Omega(x_0)/\simeq_p$ is called the \textui{fundamental group based at $x_0$}. We denote it by $\pi_1(X,x_0)$.
\end{definition}

\begin{example}
    If $Y$ is a convex subset of a topological vector space and $x_0 \in Y$, then $\pi_1(Y,x_0) = \{[e_{x_0}]\}$. Given a loop $\alpha$ based at $x_0$, then we can employ the straight-line homotopy to show that $\alpha \cong e_{x_0}$.
\end{example}

\begin{theorem}
    Suppose $X$ is a path-connected topological space. Let $p,q \in X$ and let $\gamma:I \rightarrow X$ be a path from $p$ to $q$. Then the map $\widehat{\gamma}:\pi_1(X,p) \rightarrow \pi_1(X,q)$ defined by $\widehat{\gamma}([\alpha]) = [\overline{\gamma}\ast \alpha \ast \gamma]$ is an isomorphism whose inverse is $\widehat{\overline{\gamma}}$.
\end{theorem}
    \begin{proof}
        For any $[\alpha], [\beta] \in \pi_1(X,p)$, we have:
            \begin{equation*}
            \begin{split}
                \widehat{\gamma}([\alpha] \ast [\beta])
                & = \widehat{\gamma}([\alpha \ast \beta]) \\
                & = [\overline{\gamma} \ast (a \ast b) \ast \gamma] \\
                & = [\overline{\gamma} \ast a \ast e_p \ast b \ast \gamma] \\
                & = [(\overline{\gamma} \ast a \ast \gamma )\ast (\overline{\gamma} \ast b \ast \gamma)] \\
                & = [\overline{\gamma} \ast a \ast \gamma] \ast [\overline{\gamma} \ast b \ast \gamma] \\
                & = \widehat{\gamma}([a]) \ast \widehat{\gamma}([b]).
            \end{split}
            \end{equation*}
        Hence $\widehat{\gamma}$ is a group homomorphism. Similarly, $\widehat{\overline{\gamma}}:\pi_1(X,q) \rightarrow \pi_1(X,p)$ defined by $\widehat{\overline{\gamma}}([\alpha]) = [\gamma \ast \alpha \ast \overline{\gamma}]$ is a homomorphism, and is the inverse of $\widehat{\gamma}$. Thus $\widehat{\gamma}$ is an isomorphism.
    \end{proof}

\begin{remark}
    If $X$ is path connected, by the previous theorem $\pi_1(X,x_0)$ is uniquely determined up to isomorphism. In this case, we drop the $x_0$ in the notation and write $\pi_1(X)$.
\end{remark}

\begin{remark}
    The isomorphism $\widehat{\gamma}$ depends on $\gamma$. The isomorphism is independent of the path $\gamma$ if and only if $\pi_1(X)$ is abelian (Exercise).
\end{remark}

\begin{definition}
    A topological space is \textui{simply connected} if it is path connected and its fundamental group is trivial.
\end{definition}

\begin{example}
    Any convex subset of a topological vector space is simply connected.
\end{example}

\section{Deformation Retracts and Homotopy Type}

\begin{definition}
Two continuous functions $f,g:X \rightarrow Y$ are homotopic \textui{relative to a subset} $A$ of $X$ if there exists a continuous map $F:X \times I \rightarrow Y$ such that:
    \begin{itemize}[itemsep=1pt,topsep=3pt]
        \item $F(x,0) = f(x)$ for all $x \in X$;
        \item $F(x,1) = g(x)$ for all $x \in X$;
        \item $F(a,t) = f(a)=g(a)$ for all $a \in A$ and $t \in I$.
    \end{itemize}
In this case, we say $f$ and $g$ are \textui{homotopic $\rel A$}.
\end{definition}

\begin{example}
    Let $f,g:\bfC \rightarrow \bfC$ be given by $f(z) = 0$ and $g(z) = z$ for all $z \in \bfC$. Then $f$ and $g$ are homotopic $\rel \{0\}$. The homotopy $F:\bfC \times I \rightarrow \bfC$ is given by $F(z,t) := tz$.
\end{example}

\begin{example}
    By definition, two paths $f,g:I \rightarrow X$ are path homotopic if and only if there are homotopic $\rel\{0,1\}$.
\end{example}

\begin{proposition}\label{prop:retract-isom}
    Let $(X,x_0)$ be a pointed topological space. If $f,g:X \rightarrow Y$ are continuous and homotopic $\rel\{x_0\}$, then $f_\ast = g_\ast$.
\end{proposition}
    \begin{proof}
        Let $[\alpha] \in \pi_1(X,x_0)$. It suffices to show that $f \circ \alpha \simeq_p g \circ \alpha$. By assumption, there exists a homotopy $\rel\{x_0\}$ $F:X \times I \rightarrow Y$ between $f$ and $g$. Define $H:I \times I \rightarrow Y$ by $H(s,t) := F(\alpha(s),t)$. Then:
        \begin{itemize}[itemsep=1pt,topsep=3pt]
            \item $H(s,0) = F(\alpha(s),0) = (f \circ \alpha)(s)$;
            \item $H(s,1) = F(\alpha(s),1) = (g \circ \alpha)(s)$;
            \item $H(0,t) = F(\alpha(0),t) = f(x_0) = g(x_0)$;
            \item $H(1,t) = F(\alpha(1),t) = f(x_0) = g(x_0)$;
        \end{itemize}
        Thus $f \circ \alpha \simeq_p g \circ \alpha$, hence $f_\ast = g_\ast$.
    \end{proof}

\begin{definition}
    Let $A \subset X$, $r:X \rightarrow A$ a retraction, and $j:A \rightarrow X$ the inclusion map. The subset $A$ is called a \textui{deformation retract} of $X$ if $j \circ r$ is homotopic to $\id_X \rel A$; i.e., there exists a continuous function $F:X \times I \rightarrow X$ such that:
    \begin{itemize}[itemsep=1pt,topsep=3pt]
        \item $F(x,0) = x$ for all $x \in X$;
        \item $F(x,1) = r(x)$ for all $x \in X$;
        \item $F(a,t) = a$ for all $t \in I$.
    \end{itemize}
\end{definition}

\begin{remark}
    Retracts are "instant", whereas deformation retracts happen over time.
\end{remark}

\begin{theorem}\label{thm:retract-implies-isom}
    Let $A$ be a deformation retract of $X$ and $j:A \rightarrow X$ the inclusion map. Then $j_\ast:\pi_1(A,a) \rightarrow \pi_1(X,a)$ is an isomorphism for all $a \in A$.
\end{theorem}   
    \begin{proof}
        Since $A$ is a retract of $X$, $j_\ast$ is injective by Lemma~\ref{lemma:retract-injective}. It only remains to show that $j_\ast$ is surjective. Since $A$ is a deformation retract, $j \circ r \simeq \id_X$. Hence $(j \circ r)_\ast = (\id_X)_\ast$ by Proposition~\ref{prop:retract-isom}. Since $(\id_X)_\ast$ is a bijection, it must be that $(j \circ r)_\ast = i_\ast \circ r_\ast$ is a bijection. Thus $j_\ast$ is surjective.
    \end{proof}

\begin{example}
    The space $S^1 \times B^2$ is the \textit{filled taurus}. Define $r:S^1 \times B^2 \rightarrow S^1 \times \{(0,0)\}$ by $r(x,y) := (x, (0,0))$. This is a retraction. Now let $F:(S^1 \times B^2) \times I \rightarrow S^1 \times B^2$ be given by $F((x,y),t) := (x,(1-t)y)$. Then:
        \begin{itemize}[itemsep=1pt,topsep=3pt]
            \item $F((x,y),0) = (x,y)$ for all $(x,y) \in S^1 \times B^2$,
            \item $F((x,y),1) = (x,(0,0))$ for all $(x,y) \in S^1 \times B^2$,
            \item $F\Bigl(\bigl(x,(0,0)\bigr),t\Bigr) = (x,(0,0))$ for all $t \in I$.
        \end{itemize}
    Thus $S^1 \times \{(0,0)\}$ is a deformation retract of $S^1 \times B^2$. By Theorem~\ref{thm:retract-implies-isom}, $\pi_1(S^1 \times \{(0,0)\},a) \cong \pi_1(S^1 \times B^2,a)$ for all $a \in S^1 \times \{(0,0)\}$. Since $\pi_1(S^1 \times \{(0,0)\},a) \cong \bfZ$, this means $\pi_1(S^1 \times B^2) \cong \bfZ$.
\end{example}

\section{Lifts}

\begin{definition}
	Let $p:E \rightarrow B$ be a function. If $f:X \rightarrow B$ is continuous, then a \textui{lifting} of $f$ is a map  $\tilde{f}:X \rightarrow E$ such that $p \circ f = \tilde{f}$; i.e., the following diagram commutes:
	\begin{center}
		\begin{tikzcd}
                                          & E \arrow[d, "p"] \\
			X \arrow[ru, "\tilde{f}"] \arrow[r, "f"'] & B               
		\end{tikzcd}
	\end{center}
\end{definition}

\begin{example}
	Consider $p:\bfR \rightarrow S^1$ given by $p(x) = e^{2\pi i x}$. The path $f:I \rightarrow S^1$ given by $f(s) = e^{\pi i s}$ lifts to the path $\tilde{f}:I \rightarrow R$ given by $\tilde{f}(s) = \frac{s}{2}$. Indeed, $(p \circ \tilde{f})(x) = p \left( \frac{x}{2} \right) = e^{\pi i x} = f(x)$.
\end{example}

\begin{lemma}
	Let $p:E \rightarrow B$ be a covering map. Let $e_0 \in E$ and $b_0 = p(e_0)$. Any path $f:I \rightarrow B$ beginning at $b_0$ has a unique lifting to a path $\tilde{f}:I \rightarrow E$ beginning at $e_0$.
\end{lemma}
	\begin{proof}
		For the proof of existence, we proceed by cases. Suppose $f(I)$ is contained in an elementary neighborhood $U$ of $B$. Let $V$ be the slice of $p^{-1}(U)$ containing $e_0$. Since $\restr{p}{V}$ is a homeomorphism, it is a bijection. Define $\tilde{f}:I \rightarrow E$ by $\tilde{f} = (\restr{p}{V})^{-1} \circ f$. This map is continuous as it is a composition of continuous functions. Moreover:
		\begin{align*}
			\tilde{f}(0) &= \restr{p}{V}^{-1}(f(0)) = \restr{p}{V}^{-1}(b_0) = e_0 \\
			p \circ \tilde{f} &= p \circ (\restr{p}{V}^{-1} \circ f) = f.	
		\end{align*}
		Thus $\tilde{f}$ is indeed a lift of $f$ which is path beginning at $e_0$.

		Now suppose $f(I)$ is not entirely contained in an elementary neighborhood. By the compactness of $I$ and the continuity of $f$, the set $f(I)$ is compact in $B$. Since $p$ is a covering map, the space $B$ admits a cover of elementary neighborhoods \textemdash implying that $f(I)$ admits a cover of elementary neighborhoods. By Lebesgue's Number Lemma, we can find a number $\delta$ such that any subset of $f(I)$ whose diameter is less than $\delta$ is entirely contained in an elementary neighborhood. Choose numbers $t_0,\ldots,t_n$ such that:
		\[
			0 = t_0 < t_1 < \ldots < t_{n-1} < t_n = 1
		\] 
		and $\diam([t_{i-1},t_{i}]) < \delta$. Then each $f([t_{i-1},t_{i}])$ lies in some elementary neighborhood of $B$. We can apply the previous case to each of these sets, and by the Pasting Lemma we can construct a lift $\tilde{f}$ satisfying $p \circ \tilde{f} = f$.

		{\color{red}Uniquenes\ldots}
	\end{proof}

\begin{definition}
	Let $p:E \rightarrow B$ be a covering space, $e_0 \in E$, and $b_0 = p(e_0)$. Given $[\alpha] \in \pi_1(B,b_0)$, let $\tilde{\alpha}$ be the unique lifting of $\alpha$ with $\tilde{\alpha}(0) = e_0$. The map $\Phi:\pi_1(B,b_0) \rightarrow p^{-1}(b_0)$ given by $\Phi([\alpha]) = \tilde{\alpha}(1)$ is well-defined and is called the \textui{lifting correspondence} derived from the covering map $p$.
\end{definition}

\begin{proposition}

\end{proposition}

\begin{theorem}
	$\pi_1(S^1,b_0) \cong \bfZ$.
\end{theorem}
	\begin{proof}
		Consider $p:\bfR \rightarrow S^1$ (where $S^1$ is assumed to be a subset of $\bfC$) given by $p(x) = e^{2 \pi i x}$. Let $0$ be the basepoint of $\bfR$ and $p(0) = (1,0)$ the basepoint of $S^1$. Then:
		\begin{align*}
			p^{-1}(1)
			&=  \{x \in \bfR \mid p(x) = 1\} \\
			&= \{x \in \bfR \mid e^{2 \pi i x} = 1\} \\
			&=  \{x \in \bfR \mid \cos(2 \pi x) + i \sin (2 \pi x) = 1\}\\
			& = \bfZ.
		\end{align*}	
	Since $\bfR$ is simply connected, the lifting correspondence $\Phi:\pi_1(S^1, 1) \rightarrow \bfZ$ is a bijection. It remains to show that $\Phi$ is a group homomorphism. Let $[\alpha],[\beta] \in \pi_1(S^1,1)$ and $\tilde{\alpha},\tilde{\beta}$ their respective lifts. Let $n:= \tilde{\alpha}(1)$ and $m := \tilde{\beta}(1)$. Then by definition $\Phi([\alpha]) = n$ and $\Phi([\beta]) = m$. Now define $h:I \rightarrow \bfR$ by:
	\[
		h(s) := 
		\begin{cases}
			\tilde{\alpha}(2s), & 0 \leq s \leq \frac{1}{2} \\
			\tilde{\beta}(2s-1) + n, & \frac{1}{2} \leq s \leq 1
		\end{cases}.
	\]
	Note that $h$ is continuous, and:
	\[
		(p \circ h)(s) =
		\begin{cases}
			(p \circ \tilde{\alpha})(2s), & 0\leq s \leq \frac{1}{2} \\
			p( \tilde{\beta}(2s-1) + n), & \frac{1}{2} \leq s \leq 1
		\end{cases}.
	\] 
	But note that $p(x+n) = p(x)$ for all $x \in \bfR$. Hence:
		\begin{align*}
			(p \circ h)(s) 
			&=\begin{cases}
				(p \circ \tilde{\alpha})(2s), & 0\leq s \leq \frac{1}{2},\\
				(p \circ \tilde{\beta})(2s-1), & \frac{1}{2} \leq s \leq 1
				\end{cases} \\
			&=\begin{cases}
				\alpha(2s), & 0\leq s \leq \frac{1}{2},\\
				\beta(2s-1), & \frac{1}{2} \leq s \leq 1
				\end{cases} \\
			&= (\alpha * \beta)(s).
		\end{align*}
	Since $\alpha \ast \beta$ is a path beginning at $0$, this means $h$ is its unique lift beginning at $1$. From this, we conclude that:
		\begin{align*}
			\Phi([\alpha] \ast [\beta])
			&= \Phi([\alpha \ast \beta]) \\
			&= h(1) \\
			&= n + m \\
			&= \Phi([\alpha]) + \Phi([\beta]).
		\end{align*}
	Thus $\pi_1(S^1,1) \cong \bfZ$.
\end{proof}
