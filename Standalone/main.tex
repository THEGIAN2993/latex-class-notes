\begin{theorem}
Let $X$ be a normed linear space. If $\{x_n\}_{n = 1}^\infty$ is a sequence which converges \textit{weakly} to $x$, then $\lnorm x \rnorm \leq \liminf_{n \rightarrow \infty}\lnorm x_n \rnorm$.
\end{theorem}

\begin{theorem}
Let $\{f_n\}_{n = 1}^\infty$ be a sequence in $L^+$. Suppose $f_n \rightarrow f$ a.e. Then $\int f \leq \liminf_{n \rightarrow \infty} \int f_n$.
\end{theorem}
