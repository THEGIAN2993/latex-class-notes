\begin{exercise}[manual-num={1}]
Show that if $h,h': X \rightarrow Y$ are homotopic and $k,k': Y \rightarrow Z$ are homotopic, then $k \circ h$ and $k' \circ h'$ are homotopic.
\end{exercise}
    \begin{proof}
        Since $h \cong h'$, there exists a continuous function $F:X \times I \rightarrow Y$ such that $F(x,0) = h(x)$ and $F(x,1) = h'(x)$. Since $k \cong k'$, there exists a continuous function $G:Y \times I \rightarrow Z$ such that $G(x,0) = k(x)$ and $G(x,1) = k'(x)$. Then $H:X \times I \rightarrow Z$ given by $H(x,t) := G(F(x,t),t)$ is continuous as it is the composition of continuous functions. Moreover, $H(x,0) = G(F(x,0),0) = G(h(x),0) = k(h(x))$ and $H(x,1) = G(F(x,1),1) = G(h'(x),1) = k'(h'(x))$.  Thus $h \circ k \cong h' \circ k'$.
    \end{proof}

\begin{exercise}[manual-num={2}]
Given spaces $X$ and $Y$, let $[X,Y]$ denote the set of homotopy classes of maps of $X$ into $Y$.
\begin{enumerate}[label = (\alph*),itemsep=1pt,topsep=3pt]
    \item Let $I = [0,1]$. Show that for any $X$, the set $[X,I]$ has a single element.
    \item Show that if $Y$ is path connected, the set $[I,Y]$ has a single element.
\end{enumerate}
\end{exercise}
    \begin{proof}
        (a) Let $f:X \rightarrow I$ be continuous. Let $g:X \rightarrow I$ be the constant map. Define $F:X \times I \rightarrow I$ by $F(x,t) = (1-t)f(t) + g(x)t$. Then $F$ is continuous because it is a linear combination of continuous functions. Moreover, $F(x,0) = f(x)$ and $F(x,1) = g(x)$. Therefore $f \cong g$. Since $f$ was arbitrary, $[X,I] = \{[g]\}$.

        (b) Let $f:I \rightarrow Y$ be continuous. Define $c:I \rightarrow Y$ by $c(x) = f(0)$. Since this map is constant, it is continuous. Define $F:I \times I \rightarrow Y$ by $F(x,t) = f((1-t)x)$. Then $F(x,0) = f(x)$ and $F(x,1) = f(0)$. Thus $f \cong c$.  Let $y \in Y$ be arbitrary. Since $Y$ is path connected, there is an induced homotopy from the constant map $c:I \rightarrow Y$ to the constant map $g:I \rightarrow Y$ given by $g(x) = y$. Since $f \cong c$ and $c \cong g$, we have $f \cong g$. Thus $[I,Y] = \{[g]\}$.
    \end{proof}

\begin{exercise}[manual-num={3}]
A space $X$ is said to be contractible if the identity map $i_X:X \rightarrow X$ is nullhomotopic.
\begin{enumerate}[label = (\alph*),itemsep=1pt,topsep=3pt]
    \item Show that $I$ and $\bfR$ are contractible.
    \item Show that a contractible space is path connected.
    \item Show that if $Y$ is contractible, then for any $X$, the set $[X,Y]$ has a single element.
    \item Show that if $X$ is contractible and $Y$ is path connected, then $[X,Y]$ has a single element.
\end{enumerate}
\end{exercise}
    \begin{proof}
        (a) Note that $i_{[0,1]}$ and any constant map $c:[0,1] \rightarrow [0,1]$ is continuous. Simply applying the straight-line homotopy gives $i_{[0,1]} \cong c$. An identical reasoning follows from $\bfR$ being contractible.

        (b) Let $X$ be contractible. There exists a continuous function $F:X \times I \rightarrow X$ such that $f(x,0) = i_X(x) = x$ and $f(x,1) = x_0$. Let $a,b \in X$. Define $\varphi:I \rightarrow X$ by:
            \begin{equation*}
            \begin{split}
                \varphi(t)=\begin{cases}
                f(2t,a), & 0 \leq t \leq \frac{1}{2} \\
                f(2-2t,b) & \frac{1}{2} \leq t \leq 1.
                \end{cases}
            \end{split}
            \end{equation*}
        By the pasting lemma $\varphi$ is continuous, and $\varphi(0) = a$ and $\varphi(1) = b$. Thus $X$ is path connected.

        (c) Let $f:X \rightarrow Y$ be continuous. Since $Y$ is contractible, there exists a continuous function $G:Y \times I \rightarrow Y$ such that $G(y,0) = i_Y(y) = y$ and $G(y,1) = y_0$. Define $F:X \times I \rightarrow Y \times I$ by $F(x,t) = (f(x),t)$. This map is continuous, hence the composition $G \circ F :X \times I \rightarrow Y$ is continuous. Moreover, $(G\circ F)(x,0) = f(x)$ and $(G\circ F)(x,1) = y_0$. Thus $f$ is nullhomotopic. Since $f$ was arbitrary, $[X,Y]$ contains only a single element.

        (d) Let $f:X \rightarrow Y$ be continuous. Since $X$ is contractible, there exists a continuous function $F:X \times I \rightarrow X$ such that $F(x,0) = i_X(x) = x$ and $F(X,1) = x_0$. Since $Y$ is path connected, there exists a continuous function $\varphi:I \rightarrow Y$ such that $\varphi(0) = f(x_0)$ and $\varphi(1) = y_0$. Note that $(f \circ F):X \times I \rightarrow Y$ is continuous, and $(f \circ F)(x,0) = f(x)$ and $(f \circ F)(x,1) = f(x_0)$. Now for any $y \in Y$, there exists a path from $f(x_0)$ to $y$. Since $f$ is homotopic to the constant function equal to $f(x_0)$ and since the constant function equal to $f(x_0)$ is homotopic to the constant function equal to $y_0$, by transitivity $f$ is homotopic to the constant function equal to $y_0$. Thus $[X,Y]$ contains only a single element.
    \end{proof}

\begin{exercise}
A subset $A$ of $\bfR^n$ is said to be star convex if for some point $a_0$ of $A$, all the line segments joining $a_0$ to other points of $A$ lie in $A$.
\begin{enumerate}[label = (\alph*),itemsep=1pt,topsep=3pt]
    \item Find a star convex set that is not convex.
    \item Show that if $A$ is star convex, $A$ is simply connected.
\end{enumerate}
\end{exercise}
    \begin{proof}
        (a) A typical "star" (or its higher-dimensional analogues) is star convex. Indeed, for any point in the "star" a line segment can be drawn to its center. However, the "star" is clearly not convex.

        (b) Let $A$ be a star convex. Then there exists a point $x_0 \in A$ such that for all $a \in A$, the line segment from $x_0$ to $a$ is contained in $A$. Let $f:[0,1] \rightarrow A$ be a loop based at $x_0$. Define $F:[0,1] \times [0,1] \rightarrow A$ by $F(s,t) = tx_0 + (1-t)f(s)$. This is a homotopy between $f$ and $e_{x_0}$. Thus $\pi_1(A,x_0) = \{[e_{x_0}]\}$, meaning $A$ is simply connected.
    \end{proof}

\begin{exercise}
Let $\alpha$ be a path in $X$ from $x_0$ to $x_1$; let $\beta$ be a path in $X$ from $x_1$ to $x_2$. Show that if $\gamma = \alpha \ast \beta$, then $\hat{\gamma} = \hat{\beta} \circ \hat{\alpha}$.
\end{exercise}
    \begin{proof}
        For any $[p] \in \pi_1(X,x_0)$:
            \begin{equation*}
            \begin{split}
                \hat{\beta} \left( \hat{\alpha}([p]) \right)
                & = \hat{\beta} ( [ \overline{\alpha} \ast p \ast \alpha]) \\
                & = [\overline{b} \ast (\overline{\alpha} \ast p \ast \alpha \ast \beta)] \\
                & = [(\overline{\alpha \ast \beta}) \ast p \ast (\alpha \ast \beta)] \\
                & = \hat{\gamma}([p]).
            \end{split}
            \end{equation*}
    \end{proof}

\begin{exercise}
Let $x_0$ and $x_1$ be points of the path-connect space $X$. Show that $\pi_1(X,x_0)$ is abelian if and only if for every pair $\alpha$ and $\beta$ of paths from $x_0$ to $x_1$, we have $\hat{\alpha} = \hat{\beta}$.
\end{exercise}
    \begin{proof}
        Suppose $\pi_1(X,x_0)$ is abelian. Then  $\pi_1(X,x_1)$ is abelian as well. Observe that:
            \begin{equation*}
            \begin{split}
                \hat{\alpha}([f]) 
                & = [\overline{\alpha} \ast f \ast \alpha] \\
                & = [\overline{\alpha} \ast f \ast (\beta \ast \overline{\beta}) \ast \alpha] \\
                & = [(\overline{\alpha} \ast f \ast \beta) \ast (\overline{\beta} \ast \alpha)] \\
                & = [(\overline{\beta} \ast \alpha) \ast \overline{\alpha} \ast f \ast \beta] \\
                & = [\overline{\beta} \ast f \ast \beta] \\
                & = \hat{\beta}([f]) 
            \end{split}
            \end{equation*}
        Thus $\hat{\alpha} = \hat{\beta}$.

        Conversely, let $f$ and $g$ be two lops with base point $x_0$ and let $\alpha$ be a path from $x_0$ to $x_1$. Then $\beta = f \ast \alpha$ is a path from $x_0$ to $x_1$ as well. By assumption, $\hat{\alpha} = \hat{\beta}$. Hence:
            \begin{equation*}
            \begin{split}
                [\overline{\alpha} \ast g \ast \alpha] = [\overline{\beta} \ast f \ast \beta] 
                & \iff [ \beta \overline{\alpha} \ast g \ast \alpha] = [\overline{\beta} \ast f ] \\
                & \iff [f \ast g \ast \alpha] = [g \ast \beta] \\
                & \iff [f \ast g] = [g \ast \beta \ast \overline{\alpha}] \\
                & \iff [f \ast g] = [g \ast f].
            \end{split}
            \end{equation*}
        Thus $\pi_1(X,x_0)$ is abelian.
    \end{proof}

\begin{exercise}[manual-num={5}]
    Let $A$ be a subspace of $\bfR^n$; let $h:(A,a_0) \rightarrow (Y,y_0)$. Show that if $h$ is extendable to a continuous map of $\bfR^n$ into $Y$, then $h_\ast$ is the trivial homomorphism.
\end{exercise}
    \begin{proof}
        Let $i:A \rightarrow \bfR^n$ be the inclusion. Extend $h$ to a continuous function $k:\bfR^n \rightarrow Y$. Then $h = k \circ i$, hence $h_\ast = k_\ast \circ i_\ast$ because the fundamental group is functorial. But $k_\ast$ is the trivial homomorphism because its domain is $\pi_1(\bfR^n,a_0) = \{[e_{a_0}]\}$, which is the trivial group. Thus $h_\ast = k_\ast \circ i_\ast$ is trivial.
    \end{proof}
