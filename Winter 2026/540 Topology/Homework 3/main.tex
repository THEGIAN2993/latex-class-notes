\begin{exercise}[manual-num={3}]
Let $X$ and $X'$ denote a single set in the two topologies $T$ and $T'$, respectively. Let $i:X' \rightarrow X$ be the identity function.
\begin{enumerate}[label = (\alph*),itemsep=1pt,topsep=3pt]
    \item Show that $i$ is continuous if and only if $T'$ is finer than $T$.
    \item Show that $i$ is a homeomorphism if and only if $T' = T$.
\end{enumerate}
\end{exercise}
    {\color{blue} \begin{proof}
        (a) Suppose $i$ is continuous. Then for all $U \in T$, $U = i^{-1}(U) \in T'$, hence $T \subseteq T'$. Conversely, let $T \subseteq T'$. Let $U \in T$. Then $i^{-1}(U) = U \in T \subseteq T'$. Hence $i$ is continuous.

        (b) Suppose $i$ is a homeomorphism. Then $i$ and $i^{-1}$ are continuous. Hence by part (a) $T \subseteq T'$ and $T' \subseteq T$. Thus $T = T'$. Conversely, suppose $T = T'$. Then $T \subseteq T'$ and $T' \subseteq T$. By part (a) we have that $i$ is continuous and $i^{-1}$ is continuous. Thus $i$ is a homeomorphism.
    \end{proof}}

\begin{exercise}[manual-num={10}]
Let $f:A \rightarrow B$ and $g:C \rightarrow D$ be continuous functions. Let us define a map $f \times g :A \times C \rightarrow B \times D$ by the equation $(f \times g)((a,c)) = (f(a),g(c))$. Show that $f \times g$ is continuous.
\end{exercise}
    {\color{blue} \begin{proof}
        Note that the topology on $B \times D$ is generated by the basis $\cB = \{U \times V \mid U \subseteq B \text{ is open, }V \subseteq D \text{ is open }\}$. Hence it suffices to prove that the preimage of a basis element is open in $A \times C$. Let $U \times V \in \cB$. Then $(f \times g)^{-1}(U \times V) = f^{-1}(U) \times f^{-1}(V)$. Since $f^{-1}(U)$ is open in $A$ and $f^{-1}(V)$ is open in $C$, then $f^{-1}(U) \times f^{-1}(V)$ is open in $A \times C$. Thus $f \times g$ is continuous.
    \end{proof}}

\begin{exercise}[manual-num={13}]
Let $A \subset X$; let $f:A \rightarrow Y$ be continuous; let $Y$ be Hausdorff. Show that if $f$ may be extended to a continuous function $g:\overline{A} \rightarrow Y$, then $g$ is uniquely determined by $f$.
\end{exercise}
    {\color{blue} \begin{proof}
        Suppose towards contradiction there exists continuous extensions $g,h:\overline{A} \rightarrow Y$ of $f$ and there exists $x \in \overline{A}$ such that $g(x) \neq h(x)$. Since $Y$ is Hausdorff, there exists open sets $U$ and $V$ such that $g(x) \in U$, $h(x) \in V$, and $U \cap V = \emptyset$. It follows that $x \in g^{-1}(U) \cap h^{-1}(V)$, which is an open set. But since $x \in \overline{A}$, it must be the case that $g^{-1}(U) \cap h^{-1}(V) \cap A \neq \emptyset$; i.e., there exists some $x' \in g^{-1}(U) \cap h^{-1}(V) \cap A$. Hence $g(x') \in U$, $h(x') \in V$, and $x' \in A$. But since $g$ and $h$ are extensions of $f$, we have that $g(x') = h(x')$. Hence $U \cap V \neq \emptyset$, which contradicts $Y$ being Hausdorff. Thus any continuous extension of $f$ is unique.
    \end{proof}}

\begin{exercise}[manual-num={1(a)}]
In $\bfR^n$, define:
    \begin{equation*}
    \begin{split}
        d'(x,y) = |x_1 - y_1| + ... + |x_n - y_n|.
    \end{split}
    \end{equation*}
Show that $d'$ is a metric that induces the usual topology of $\bfR^n$. Sketch the basis elements under $d'$ when $n = 2$.
\end{exercise}
    {\color{blue} \begin{proof}
        We will first show that $d'$ is a metric. Note that:
            \begin{equation*}
            \begin{split}
                d'(x,y) 
                & = |x_1 - y_1| + ... + |x_n - y_n| \\
                & = |y_1 - x_1| + ... + |y_n - x_n| \\
                & = d'(y,x).
            \end{split}
            \end{equation*}
        Similarly, if $d'(x,y) = 0$, then $|x_1 - y_1|+...+|x_n - y_n| = 0$. Hence $|x_i - y_i| = 0$ for each $i =1,2,...,n$. Thus $x_i = y_i$ for all $i=1,2,...,n$. Hence $x = y$. Lastly, observe that:
            \begin{equation*}
            \begin{split}
                d'(x,y) 
                & = |x_1 - y_1| + ... + |x_n - y_n| \\
                & \leq |x_1 - z_1| + |z_1 - y_1| + ... + |x_n - z_n| + |z_n - y_n| \\
                & = \left( |x_1 - z_1| + ... + |x_n - z_n| \right) + \left( |z_1 - y_1| + ... + |z_n - y_n| \right) \\
                & = d'(x,z) + d'(z,y).
            \end{split}
            \end{equation*}
        Thus $d'$ is a metric on $\bfR^n$.

        We will now show that $d'$ induces the standard topology on $\bfR^n$. Note that the standard topology on $\bfR^n$ is induced by the metric $d^2(x,y) = \left(\sum_{i = 1}^n |x_i - y_i|^2\right)^\frac{1}{2}$. Observe that:
            \begin{equation*}
            \begin{split}
                d^2(x,y) 
                & = \left(\sum_{i = 1}^n |x_i - y_i|^2\right)^\frac{1}{2} \\
                & \leq \sum_{i = 1}^n \left( |x_i - y_i|^2 \right)^\frac{1}{2} \\
                & = \sum_{i = 1}^n |x_i - y_i| \\
                & = d'(x,y).
            \end{split}
            \end{equation*}
        Let $x \in \bfR^n$ and $\epsilon > 0$ be arbitrary. If $y \in B_{d'}(x,\epsilon)$, then $d(x,y) < \epsilon$. Hence $d^2(x,y) < \epsilon$, which means $y \in B_{d^2}(x,\epsilon)$. Thus $B_{d'}(x,\epsilon) \subseteq B_{d^2}(x,\epsilon)$. Now observe that:
            \begin{equation*}
            \begin{split}
                d'(x,y) 
                & = \sum_{i = 1}^n |x_i - y_i| \\
                & \leq n \cdot \max_{i = 1}^n |x_i - y_i| \\
                & = n \cdot \left( (\max_{i = 1}^n |x_i - y_i|)^2 \right)^\frac{1}{2} \\
                & \leq n \cdot \left( \sum_{i = 1}^n |x_i - y_i|^2  \right)^\frac{1}{2} \\
                & = n \cdot d^2(x,y).
            \end{split}
            \end{equation*}

        Let $x \in \bfR^n$ and $\epsilon > 0$ be arbitary. If $y \in B_{d^2}(x,\frac{\epsilon}{n})$, then $n \cdot d^2(x,y) < \epsilon$. Hence $d'(x,y) < \epsilon$, which means $y \in B_{d'}(x,\epsilon)$. Thus $B_{d^2}(x,\frac{\epsilon}{n}) \subseteq B_{d'}(x,\epsilon)$. Therefore the topology induced by $d'$ is the same by the topology induced by $d^2$, which is the standard topology on $\bfR^n$.

    \end{proof}}

\begin{exercise}[manual-num={3}]
Let $X$ be a metric space with metric $d$.
\begin{enumerate}[label = (\alph*),itemsep=1pt,topsep=3pt]
    \item Show that $d:X \times X \rightarrow \bfR$ is continuous.
    \item Let $X'$ denote a space having the same underlying set as $X$. Show that if $d:X' \times X' \rightarrow \bfR$ is continuous, then the topology of $X'$ is finer than the topology of $X$.
\end{enumerate}
\end{exercise}
    {\color{blue} \begin{proof}
        (a) Equip $X \times X$ with the product metric:
            \begin{equation*}
            \begin{split}
                D((x_1,y_1),(x_2,y_2)) = d(x_1,y_1) + d(x_2,y_2).
            \end{split}
            \end{equation*}
        Let $(x,y), (x',y') \in X \times X$. By the triangle inequality:
            \begin{equation*}
            \begin{split}
                d(x,y) \leq d(x,x') + d(x',y') + d(y',y).
            \end{split}
            \end{equation*}
        Hence $d(x,y) - d(x',y') \leq d(x,x') + d(y,y')$. Applying the triangle inequality similarly to $d(x',y')$ gives the inequality $d(x',y') - d(x,y) \leq d(x,x') + d(y,y')$. Hence:
            \begin{equation*}
            \begin{split}
                |d(x,y) - d(x',y')| 
                & \leq d(x,x') + d(y,y') \\
                & = D((x,y),(x',y')).
            \end{split}
            \end{equation*}
        Thus $d$ is Lipschitz, establishing $d$ as continuous.
    \end{proof}}

\begin{exercise}[manual-num={1}]
Let $A \subset X$. If $d$ is a metric for the topology of $X$, show that $\restr{d}{A \times A}$ is a metric for the subspace topology on $A$.
\end{exercise}
    {\color{blue} \begin{proof}
        Clearly the restriction of $d$ onto $A \times A$ is a metric. Let $a \in A \cap U$, where $U \subseteq X$ is open. Then there exists some $r > 0$ such that $B_d(a,r) \subseteq U$. Hence $A \cap B_d(a,r) \subseteq A \cap U$. But $A \cap B_d(a,r) = B_{\restr{d}{A \times A}}(a,r)$, hence $B_{\restr{d}{A \times A}}(a,r) \subseteq A \cap U$. Thus $\restr{d}{A \times A}$ is a metric for the subspace topology on $A$.
    \end{proof}}

\begin{exercise}[manual-num={2}]
Let $X$ and $Y$ be metric spaces with metrics $d_X$ and $d_Y$, respectively. Let $f:X \rightarrow Y$ have the property that for every pair of points $x_1,x_2$ of $X$:
    \begin{equation*}
    \begin{split}
        d_Y(f(x_1),f(x_2)) = d_X(x_1,x_2).
    \end{split}
    \end{equation*}
Show that $f$ is an imbedding. (The above condition is called an isometry)
\end{exercise}
    {\color{blue} \begin{proof}
        We must first show that $f$ is continuous. Let $x_0 \in X$ and $\epsilon > 0$. Pick $\delta = \epsilon$. Then for any $x \in X$, $d_X(x,x_0) < \delta$ implies $d_Y(f(x),f(x_0)) = d_X(x,x_0) < \delta = \epsilon$.

        Next, we must show that $f$ is injective. Suppose $f(x) = f(x')$. Then $0 = d_Y(f(x),f(x')) = d_X(x,x')$. Hence $x = x'$, establishing that $f$ is injective.

        Since $f:X \rightarrow Y$ is continuous, $g:X \rightarrow f(X)$ is also continuous. In particular, since $f$ is injective, $g$ is bijective. We will now show that $g^{-1}$ is an isometry. Observe that:
            \begin{equation*}
            \begin{split}
                d_X(g^{-1}(y_0),g^{-1}(y)) 
                & = d_X(g^{-1}(g(x_0)),g^{-1}(g(x))) \\
                & = d_X(x_0,x) \\
                & = d_Y(g(x_0),g(x)) \\
                & = D_Y(y_0,y).
            \end{split}
            \end{equation*}
        Since $g^{-1}$ is an isometry, it is continuous. Thus $g:X \rightarrow f(X)$ is a homeomorphism, establishing $f$ as an imbedding.
    \end{proof}}

\begin{exercise}[manual-num={10}]
Using the closed set formulation of continuity, show that the following are closed subsets of $\bfR^2$:
    \begin{equation*}
    \begin{split}
        A &= \{(x,y) \mid xy = 1\}, \\
        S^1 &= \{(x,y) \mid x^2 + y^2 = 1\}, \\
        B^2 &= \{(x,y) \mid x^2 + y^2 \leq 1\}.
    \end{split}
    \end{equation*}
\end{exercise}
    {\color{blue} \begin{proof}
        Define $f_1:\bfR^2 \rightarrow \bfR$ by $(x,y) \mapsto xy$. This map is continuous, and:
            \begin{equation*}
            \begin{split}
                f_1^{-1}(\{1\})
                & = \{(x,y) \mid f((x,y)) \in \{1\}\} \\
                & = \{(x,y) \mid xy = 1\} \\
                & = A.
            \end{split}
            \end{equation*}
        Since $\{1\}$ is closed in $\bfR$, $A$ is closed in $\bfR^2$.

        Define $f_2:\bfR^2 \rightarrow \bfR$ by $(x,y) \mapsto x^2 + y^2$. This map is continuous, and:
            \begin{equation*}
            \begin{split}
                f_2^{-1}(\{1\})
                & = \{(x,y) \mid f((x,y)) \in \{1\}\} \\
                & = \{(x,y) \mid x^2 + y^2 = 1\} \\
                & = S^1.
            \end{split}
            \end{equation*}
        Since $\{1\}$ is closed in $\bfR$, $S^1$ is closed in $\bfR^2$. Moreover:
            \begin{equation*}
            \begin{split}
                f_2^{-1}((-\infty,1]) 
                & = \{(x,y) \mid f((x,y)) \in (-\infty,1]\} \\
                & = \{(x,y) \mid x^2 + y^2 \leq 1\} \\
                & = B^2.
            \end{split}
            \end{equation*}
        Since $(-\infty,1]$ is closed in $\bfR$, $B^2$ is closed in $\bfR^2$.
    \end{proof}}
