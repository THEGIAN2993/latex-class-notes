\begin{exercise}[manual-num={1}]
\phantom{a}
\begin{enumerate}[label = (\alph*),itemsep=1pt,topsep=3pt]
    \item Let $T$ and $T'$ be two topologies on the set $X$; suppose that $T' \supset T$. What does compactness of $X$ under one of these topologies imply about compactness under the other?
    \item Show that if $X$ is compact Hausdorff under both $T$ and $T'$, then either $T$ and $T'$ are equal or they are not comparable.
\end{enumerate}
\end{exercise}
    \begin{proof}
        (a) If $X$ is compact under $T'$, then it is compact under $T$. Indeed, if $\cU$ is an open cover of $X$ in $T$, then $\cU$ is also an open cover of $X$ in $T'$. Since $X$ is compact under $T'$, then $\cU$ contains a finite subcover of $X$. Since our choice of $\cU$ was arbitrary, $X$ must be compact under $T$. However, if $X$ is compact under $T$, we do not necessarily have compactness under $T'$. Consider $\bfR$ with the discrete topology and the trivial topology. The trivial topology is courser than the discrete topology, and $\bfR$ is compact under the trivial topology, however $\bfR$ is \textit{not} compact under the discrete topology. Indeed, $\{\{x\}| x \in X\}$ is an open cover of $X$ which admits no finite subcover.

        (b) Suppose $T \subseteq T'$. Consider the identity map $I:(X,T') \rightarrow (X,T)$. Since it is a map from a compact space into a Hausdorff space, $I$ must be a homeomorphism. Therefore $T = T'$.
    \end{proof}

\begin{exercise}[manual-num={2(a)}]
Show that in the finite complement topology on $\bfR$, every subspace is compact.
\end{exercise}
    \begin{proof}
        Let $A \subseteq \bfR$. Suppose $\{U_i\}_{i \in I}$ is an open cover of $A$. Note that the complement of each open cover will be a finite set. Consider the set $A \setminus U_j$. If $A \setminus U_j = \emptyset$, then $U_j$ completely contains $A$, hence it is a finite subcover of $\{U_i\}_{i \in I}$. If $A \setminus U_j \neq \emptyset$, then $A \setminus U_j$ must be finite, as $A \setminus U_j = A \cap U_j^c$. Let $\{x_1,...,x_n\} := A \setminus U_j$. Since each $x_1,...,x_n \in A \subseteq \bigcup_{i \in I}U_i$, we can find $U_1,...,U_n \in \{U_i\}_{i \in I}$ such that $x_j \in U_j$ for each $j=1,...,n$. Then by construction $A \cap \bigcap_{j = 1}^n U_j^c = \emptyset$, hence $A \subseteq \bigcup_{j = 1}^n U_j$. Thus $A$ is compact.
    \end{proof}

\begin{exercise}[manual-num={5}]
Let $A$ and $B$ be disjoint compact subspaces of the Hausdorff space $X$. Show that there exist disjoint open sets $U$ and $V$ containing $A$ and $B$, respectively.
\end{exercise}
    \begin{proof}
        For each $b \in B$, there exists opens sets $U_b, V_b$ such that $b \in U_b$, $A \subseteq V_b$, and $U_b \cap V_b = \emptyset$. Consider the cover $B \subseteq \bigcup_{b \in B}U_b$. Since $B$ is compact, there exists a finite subcover such that $B \subseteq \bigcup_{i = 1}^n U_{b_i}$. Since $A \subseteq V_{b_i}$ for each $b_i$, then $A \subseteq \bigcap_{i = 1}^n V_{b_i}$. All together, we've found open sets $\bigcup_{i = 1}^n U_{b_i}$, $\bigcap_{i = 1}^n V_{b_i}$ such that $A \subseteq \bigcap_{i = 1}^n V_{b_i}$, $B \subseteq \bigcup_{i = 1}^n U_{b_i}$, and $\left(\bigcap_{i = 1}^n V_{b_i}\right) \cap \left(\bigcup_{i = 1}^n U_{b_i}\right) = \emptyset$.
    \end{proof}

\begin{exercise}[manual-num={6}]
Show that if $f:X \rightarrow Y$ is continuous, where $X$ is compact and $Y$ is Hausdorff, then $f$ is a closed map.
\end{exercise}
    \begin{proof}
        Let $C$ be a closed set in $X$. Then $C$ is compact. Since $f$ is continuous, $f(C)$ is compact. Since $Y$ is Hausdorff, $f(C)$ is closed. Thus $f$ is a closed map.
    \end{proof}

\begin{exercise}[manual-num={7}]
Show that if $Y$ is compact, then the projection $\pi_1:X \times Y \rightarrow X$ is a closed map.
\end{exercise}
    \begin{proof}
        Let $C$ be a closed subset of $X \times Y$. Let $x \in X \setminus \pi_1(C)$. Then the slice $\{x\} \times Y$ is disjoint from $C$. Because $Y$ is compact, by the Tube Lemma, there is a neighborhood $W$ of $x$ such that the whole tube $W \times Y$ is disjoint from $C$. Therefore, $W$ is a neighborhood of $x$ which is disjoint from $\pi_1(C)$. Hence $X \setminus \pi_1(C)$ is open, which means $\pi_1(C)$ is closed. 
    \end{proof}

\begin{exercise}[manual-num={2}]
Let $X$ be a metric space with metric $d$; let $A \subset X$ be nonempty.
    \begin{enumerate}[label = (\alph*),itemsep=1pt,topsep=3pt]
        \item Show that $d(x,A) = 0$ if and only if $x \in \overline{A}$.
        \item Show that if $A$ is compact, $d(x,A) = d(x,a)$ for some $a \in A$.
    \end{enumerate}
\end{exercise}
    \begin{proof}
        (a) Let $U \subseteq X$ be an open set. Suppose $d(x,A) = 0$. We can find a sequence $(a_n)_n \subset A$ such that $d(x,a_n) \rightarrow 0$. Hence $a_n \rightarrow x$. Find $N \in \bfN$ sufficiently large such that $a_N \in U$. Hence $U \cap A \neq \emptyset$. Since $U$ was arbitrary, $x \in \overline{A}$. Conversely, let $x \in \overline{A}$ and $\epsilon >0$. We can find a sequence $(a_n)_n \subset A$ such that $a_n \rightarrow x$. Find $N \in \bfN$ sufficiently large such that $d(x, a_N) < \epsilon$. Then $\inf\{d(x,a) \mid a \in A\} \leq d(x,a_N) < \epsilon$, hence $d(x,A) < \epsilon$. Since $\epsilon > 0$ was arbitrary, $d(x,A) = 0$.

        (b) Let $x \in X$. Consider the map $f:A \rightarrow X$ given by $f(a) = d(x,a)$. Then $f$ is continuous, and $f(A) = \{d(x,a) \mid a \in A\}$. By the Extreme Value Theorem, there exists some $a' \in A$ such that $d(x,a') = \inf\{d(x,a) \mid a \in A\} = d(x,A)$.
    \end{proof}

\begin{exercise}[manual-num={3}]
Recall that $\bfR_K$ denoted $\bfR$ in the $K$-topology.
\begin{enumerate}[label = (\alph*),itemsep=1pt,topsep=3pt]
    \item Show that $[0,1]$ is not compact as a subspace of $\bfR_K$.
    \item Show that $\bfR_K$ is connected. (Hint $(-\infty,0)$ and $(0,\infty)$ inherit their usual topologies as subspaces of $\bfR_K$)
    \item Show that $\bfR_K$ is not path connected.
\end{enumerate}
\end{exercise}
    \begin{proof}
        (a) Recall that the $K$-topology is generated by the basis of open intervals $(a,b)$ together with all sets of the form $(a,b) \setminus K$. Define:
            \begin{equation*}
            \begin{split}
                U_n := \Bigl({\ts\frac{1}{n}} ,2\Bigr) \cup \Bigl((-1,1)\setminus K\Bigr)\h1, \h3 n \in \bfN.
            \end{split}
            \end{equation*}
        Then $\{U_n\}_{n \in \bfN}$ is an open cover which does not admit a finite subcover.

        (b) Since $(-\infty,0)$ and $(0,\infty)$ inherit their usual topologies as subspaces of $\bfR_K$, they are connected. Hence their closures are connected; i.e., the sets $(-\infty,0]$ and $[0,\infty)$ are connected. Since their intersections are nonempty, their union is connected as well, which equals $\bfR_K$.

    \end{proof}

\begin{exercise}[manual-num={4}]
Show that a connected metric space having more than one point is uncountable.
\end{exercise}
    \begin{proof}
        Let $(X,d)$ be a connected metric space. Let $x_0,x_1 \in X$. Define $f:X \rightarrow \bfR$ by:
            \begin{equation*}
            \begin{split}
                f(x) = \frac{d(x_0,x)}{d(x_0,x) + d(x_1,x)}.
            \end{split}
            \end{equation*}
        Clearly $f$ is continuous, and $f(x_0) = 0$ and $f(x_1) = 1$. Since the continuous image of a connected space is connected, it must be the case that $f(X) = [0,1]$. Since $[0,1]$ is uncountable, $X$ is uncountable.
    \end{proof}
