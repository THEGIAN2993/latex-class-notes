\usepackage{datetime}

\usepackage{xparse}
\usepackage{fancyhdr}

\pagestyle{fancy}
\fancyhf{}
\cfoot{\scriptsize\thepage}
\fancyhead[R]{\scalebox{0.8}{\nouppercase{\rightmark}}}
\fancyhead[L]{\scalebox{0.8}{\nouppercase{\leftmark}}}
\renewcommand{\headrulewidth}{0.4pt}

\newcommand{\currentparttitle}{}
\newcommand{\currentchaptertitle}{}
\providecommand{\projecttitle}{}

\let\oldpart\part
\RenewDocumentCommand{\part}{s o m}{%
    \gdef\currentparttitle{#3}%
    \IfBooleanTF{#1}
        {\oldpart*{#3}}
        {\IfNoValueTF{#2}{\oldpart{#3}}{\oldpart[#2]{#3}}}%
}

\let\oldchapter\chapter
\RenewDocumentCommand{\chapter}{s o m}{%
    \gdef\currentchaptertitle{#3}%
    \IfBooleanTF{#1}
        {\oldchapter*{#3}}
        {\IfNoValueTF{#2}{\oldchapter{#3}}{\oldchapter[#2]{#3}}}%
}

\newcommand{\plainheadertext}{%
    \if\relax\detokenize\expandafter{\currentparttitle}\relax
        \currentchaptertitle
    \else
        \currentparttitle
    \fi
}

\fancypagestyle{plain}{%
    \fancyhf{}
    \cfoot{\scriptsize\thepage}
    \fancyhead[L]{\scalebox{0.8}{\nouppercase{\plainheadertext}}}
    \fancyhead[R]{\scalebox{0.8}{\projecttitle}}
    \renewcommand{\headrulewidth}{0.4pt}
}
\aliaspagestyle{part}{empty}

    	inherit-style=definition,
    	spaceabove=1.5em,
    	spacebelow=1.5em,

    	spaceabove=1.5em,
    	spacebelow=1.5em,

\usepackage{enumitem}
\usepackage[explicit]{titlesec}
\usepackage{appendix}

\titleformat{\chapter}[display]
    	{\bfseries\LARGE\raggedright}
    	{Chapter {\thechapter}}
    	{.1ex minus .1ex}
    	{\Large #1}

\titlespacing{\chapter}
    	{3pc}{*3}{40pt}[3pc]

\titleformat{\section}[block]
    	{\normalfont\bfseries\large}
    	{}{0pt}
    	{\fbox{\strut \S\ \thesection.\ #1}}

\titlespacing{\section}
    	{0pt}{3ex plus .1ex minus .2ex}{3ex plus .1ex minus .2ex}

\setlength{\fboxsep}{3pt}    % padding inside the box
\setlength{\fboxrule}{0.6pt} % border thickness

\titleformat{name=\section,numberless}[block]
    	{\normalfont\bfseries\large\filcenter}
    	{}{0pt}
   	 {#1}

	 
\setcounter{secnumdepth}{2} % number subsections
\setcounter{tocdepth}{2}    % show subsections in table of contents

\titleformat{\subsection}[block]
    {\normalfont\bfseries\large}
    {}{0pt}
    {\Roman{subsection}.\ #1}
